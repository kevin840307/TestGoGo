--SKC25J105
#include "txt_25105.txt"

function init_data()
	skillflag 	= "25105"
	TXT_SkillName()
	boost 		= "EffMAP"
end

function start()
	init_data()
	consume 	= "MP"
	varyDelta 	= 20

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 25105
	
	defender 	= GetCallerPtr()	
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 9
	elseif lv == 2 then
		return 11
	elseif lv == 3 then
		return 13
	elseif lv == 4 then
		return 16
	else
		return 19
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 72
	elseif lv == 2 then
		return 90
	elseif lv == 3 then
		return 112
	elseif lv == 4 then
		return 135
	else
		return 162
	end
end
