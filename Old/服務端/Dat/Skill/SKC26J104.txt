function start()
	MinHit=2
	MinHitDelta=3
	BlockByShield = 0
	consume = "MP"
--	conTotal = 20
	APcount = "MAP"
	DPcount = "MDP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	
	conTotal = GetConTotal(skillLv)
		
	Attack_Range(125)
end

function GetPara(lv)
	if lv == 1 then
		return 74
	elseif lv == 2 then
		return 100
	elseif lv == 3 then
		return 134
	elseif lv == 4 then
		return 170
	else
		return 214
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 33
	elseif lv == 2 then
		return 42
	elseif lv == 3 then
		return 54
	elseif lv == 4 then
		return 69
	else
		return 84
	end
end
