--SKC29J106
#include "txt_29106.txt"

function init_data()
	skillflag 		= "29106"
	TXT_SkillName()
	boost			= "EffAP"
	other_boost 	= "EffMAP"
end

function start()
	init_data()
	consume 		= "MP"
	
	varyDelta 		= 10

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 29106
	
	defender 		= GetCallerPtr()
	
	skillLv 		= GetSkillLv()
	effect			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()	
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 11
	elseif lv == 3 then
		return 14
	elseif lv == 4 then
		return 18
	else
		return 21
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 55
	elseif lv == 2 then
		return 72
	elseif lv == 3 then
		return 90
	elseif lv == 4 then
		return 112
	else
		return 135
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -8
	elseif lv == 2 then
		return -11
	elseif lv == 3 then
		return -14
	elseif lv == 4 then
		return -18
	else
		return -21
	end
end
