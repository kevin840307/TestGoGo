#include "txt_27105.txt"

function init_data()
	skillflag 	= "27105"
	TXT_SkillName()
	boost		= "EffAP"
end

function start()
	init_data()
	consume 	= "MP"
	varyDelta 	= 10

	recastTime 	= 600
	duration 	= 600
	
	skillid 	= 27105
	
	defender	= GetTargetPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 5
	elseif lv == 2 then
		return 6
	elseif lv == 3 then
		return 9
	elseif lv == 4 then
		return 11
	else
		return 13
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 21
	elseif lv == 2 then
		return 28
	elseif lv == 3 then
		return 36
	elseif lv == 4 then
		return 45
	else
		return 56
	end
end
