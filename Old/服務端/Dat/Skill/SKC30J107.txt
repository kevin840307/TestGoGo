function start()
	MinHit=2
	MinHitDelta=3
	BlockByShield = 0
	consume = "Will"
--	conTotal = 20
	APcount = "MAP"
	DPcount = "MDP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	
	conTotal = GetConTotal(skillLv)
		
	Attack()
end

function GetPara(lv)
	if lv == 1 then
		return 340
	elseif lv == 2 then
		return 520
	elseif lv == 3 then
		return 740
	elseif lv == 4 then
		return 1000
	else
		return 130
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 26
	elseif lv == 2 then
		return 35
	elseif lv == 3 then
		return 46
	elseif lv == 4 then
		return 59
	else
		return 74
	end
end
