--SKC27J103
#include "txt_27103.txt"

function init_data()
	skillflag 	= "27103"
	TXT_SkillName()
	boost		= "EffSpeed"
end

function start()
	init_data()
	consume 	= "Will"
	
	varyDelta 	= 10

	recastTime = GetDuration(skillLv)
	duration 	= recastTime
	
	skillid 	= 27103
	
	defender	= GetTargetPtr()

	skillLv 	= GetSkillLv()	
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetDuration(lv)
	if lv == 1 then
		return 120
	elseif lv == 2 then
		return 240
	else
		return 480
	end
end

function GetPara(lv)
	return 1
end

function GetConTotal(lv)
	if lv == 1 then
		return 14
	elseif lv == 2 then
		return 26
	else
		return 46
	end
end
