#include "txt_23101.txt"

function init_data()
	skillflag 	= "23101"
	TXT_SkillName()
	boost 		= "EffAP"
end

function start()
	init_data()
	consume 	= "MP"
	varyDelta 	= 20

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 23101	
	defender 	= GetCallerPtr()
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 3
	elseif lv == 2 then
		return 4
	elseif lv == 3 then
		return 5
	elseif lv == 4 then
		return 6
	else
		return 9
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 22
	elseif lv == 2 then
		return 30
	elseif lv == 3 then
		return 42
	elseif lv == 4 then
		return 55
	else
		return 72
	end
end
