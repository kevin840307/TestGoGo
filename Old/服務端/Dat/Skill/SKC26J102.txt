#include "txt_26102.txt"

function init_data()
	skillflag 	= "26102"
	TXT_SkillName()
	boost 		= "EffMAP"
end

function start()
	init_data()
	consume 	= "MP"
	varyDelta 	= 20

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 26102
	
	defender 	= GetCallerPtr()	
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 3
	elseif lv == 2 then
		return 4
	elseif lv == 3 then
		return 5
	elseif lv == 4 then
		return 6
	else
		return 9
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 11
	elseif lv == 2 then
		return 15
	elseif lv == 3 then
		return 21
	elseif lv == 4 then
		return 28
	else
		return 36
	end
end
