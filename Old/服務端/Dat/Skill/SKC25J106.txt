#include "txt_25106.txt"

function start()
	consume 	= "Will"
	boost 		= "HP"
	varyDelta 	= 10

	skillid 	= 25106
	TXT_SkillName()
	skillflag 	= "25106"
	
	defender 	= GetCallerPtr()	
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	Cure()
end

function GetPara(lv)
	if lv == 1 then
		return 156
	elseif lv == 2 then
		return 234
	elseif lv == 3 then
		return 329
	elseif lv == 4 then
		return 441
	else
		return 569
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 26
	elseif lv == 2 then
		return 35
	elseif lv == 3 then
		return 48
	elseif lv == 4 then
		return 59
	else
		return 74
	end
end
