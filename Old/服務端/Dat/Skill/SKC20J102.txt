function start()

	MinHit=2
	MinHitDelta=3
	BlockByShield = 1
	consume = "MP"
	--conTotal = 20

	APcount = "AP"
	DPcount = "DP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	conTotal = GetConTotal(skillLv)
		
	Attack()
end

function GetPara(lv,skill_type)
	if lv == 1 then
		return 4
	elseif lv == 2 then
		return 5
	elseif lv == 3 then
		return 9
	elseif lv == 4 then
		return 13
	else
		return 19
	end		
end

function GetConTotal(lv)
	if lv == 1 then
		return 6
	elseif lv == 2 then
		return 8
	elseif lv == 3 then
		return 11
	elseif lv == 4 then
		return 15
	else
		return 21
	end			
end
