#include "txt_27106.txt"

function init_data()
	skillflag 	= "27106"
	TXT_SkillName()
	boost		= "EffMAP"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 600
	duration 	= 600
	
	skillid 	= 27106
	
	defender	= GetTargetPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 6
	elseif lv == 2 then
		return 9
	elseif lv == 3 then
		return 11
	elseif lv == 4 then
		return 13
	else
		return 16
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 28
	elseif lv == 2 then
		return 26
	elseif lv == 3 then
		return 45
	elseif lv == 4 then
		return 56
	else
		return 68
	end
end
