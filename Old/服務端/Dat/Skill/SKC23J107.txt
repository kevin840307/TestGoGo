--SKC23J107
#include "txt_23107.txt"

function init_data()
	skillflag 		= "23107"
	TXT_SkillName()
	boost 			= "EffHR"
	other_boost 	= "EffAP"
end

function start()
	init_data()
	consume 		= "MP"

	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 23107
	
	defender 		= GetCallerPtr()
	
	--attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 9
	elseif lv == 3 then
		return 10
	elseif lv == 4 then
		return 11
	else
		return 12
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 112
	elseif lv == 2 then
		return 135
	elseif lv == 3 then
		return 162
	elseif lv == 4 then
		return 190
	else
		return 222
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -18
	elseif lv == 2 then
		return -21
	elseif lv == 3 then
		return -26
	elseif lv == 4 then
		return -30
	else
		return -35
	end
end
