function start()

	MinHit=2
	MinHitDelta=3
	BlockByShield = 0
	consume = "MP"
--	conTotal = 20
	APcount = "MAP"
	DPcount = "MDP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	
	conTotal = GetConTotal(skillLv)
		
	Attack()
end

function GetPara(lv)
	if lv == 1 then
		return 88
	elseif lv == 2 then
		return 130
	elseif lv == 3 then
		return 175
	elseif lv == 4 then
		return 235
	else
		return 375
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 14
	elseif lv == 2 then
		return 19
	elseif lv == 3 then
		return 24
	elseif lv == 4 then
		return 30
	else
		return 38
	end
end
