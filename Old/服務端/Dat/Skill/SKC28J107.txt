--SKC28J107
#include "txt_28107.txt"

function start()
	consume 	= "Will"
	boost		= "HP"
	other_boost = "MP"
	
	varyDelta 	= 10

	skillid 	= 28107
	TXT_SkillName()
	skillflag 	= "28107"
	
	defender	= GetTargetPtr()
	if defender == 0 then
		defender = GetCallerPtr()
	end
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	other_effect = GetPara_Other(skillLv)
	
	Cure_Range_2_Boost(400)
end

function GetPara(lv)
	if lv == 1 then
		return 170
	elseif lv == 2 then
		return 260
	elseif lv == 3 then
		return 370
	elseif lv == 4 then
		return 500
	else
		return 650
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return 90
	elseif lv == 2 then
		return 135
	elseif lv == 3 then
		return 190
	elseif lv == 4 then
		return 255
	else
		return 330
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 32
	elseif lv == 2 then
		return 70
	elseif lv == 3 then
		return 92
	elseif lv == 4 then
		return 118
	else
		return 148
	end
end
