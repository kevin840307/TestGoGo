#include "txt_30102.txt"

function init_data()
	skillflag 	= "30102"
	TXT_SkillName()
	boost		= "EffDR"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 30102
	
	defender 	= GetCallerPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)

	GoBoost()	
end

function GetPara(lv)
	if lv == 1 then
		return 3
	elseif lv == 2 then
		return 7
	else
		return 10
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 11
	elseif lv == 2 then
		return 36
	else
		return 81
	end
end
