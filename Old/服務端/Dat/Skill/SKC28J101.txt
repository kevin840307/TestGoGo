#include "txt_28101.txt"

function start()
	consume 	= "MP"
	boost		= "HP"
	
	varyDelta 	= 10

	skillid 	= 28101
	TXT_SkillName()
	skillflag 	= "28101"
	
	defender	= GetTargetPtr()
	if defender == 0 then
		defender = GetCallerPtr()
	end
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	Cure()
end

function GetPara(lv)
	if lv == 1 then
		return 11
	elseif lv == 2 then
		return 15
	elseif lv == 3 then
		return 21
	elseif lv == 4 then
		return 28
	else
		return 36
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 11
	elseif lv == 2 then
		return 15
	elseif lv == 3 then
		return 21
	elseif lv == 4 then
		return 28
	else
		return 36
	end
end
