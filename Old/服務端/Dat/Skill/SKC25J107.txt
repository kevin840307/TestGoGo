--SKC25J107
#include "txt_25107.txt"

function init_data()
	skillflag 		= "25107"
	TXT_SkillName()
	boost 			= "EffMPMAX"
	other_boost 	= "EffMDP"
end

function start()
	init_data()
	consume 		= "MP"
	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 25107
	
	defender 		= GetCallerPtr()	
	
	attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 90
	elseif lv == 2 then
		return 108
	elseif lv == 3 then
		return 130
	elseif lv == 4 then
		return 152
	else
		return 178
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return 5
	elseif lv == 2 then
		return 6
	elseif lv == 3 then
		return 7
	elseif lv == 4 then
		return 8
	else
		return 9
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 112
	elseif lv == 2 then
		return 135
	elseif lv == 3 then
		return 162
	elseif lv == 4 then
		return 190
	else
		return 222
	end
end
