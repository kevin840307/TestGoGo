#include "txt_29102.txt"

function init_data()
	skillflag 	= "29102"
	TXT_SkillName()
	boost		= "EffHR"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 29102
	
	defender 	= GetCallerPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)

	GoBoost()	
end

function GetPara(lv)
	if lv == 1 then
		return 2
	elseif lv == 2 then
		return 4
	elseif lv == 3 then
		return 6
	elseif lv == 4 then
		return 8
	else
		return 10
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 15
	elseif lv == 2 then
		return 30
	elseif lv == 3 then
		return 55
	elseif lv == 4 then
		return 90
	else
		return 135
	end
end
