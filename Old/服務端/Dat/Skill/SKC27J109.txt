#include "txt_27109.txt"

function init_data()
	skillflag 	= "27109"
	TXT_SkillName()
	boost		= "EffHR"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 600
	duration 	= 600
	
	skillid 	= 27109
	
	defender	= GetTargetPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 9
	elseif lv == 3 then
		return 10
	elseif lv == 4 then
		return 11
	else
		return 12
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 56
	elseif lv == 2 then
		return 68
	elseif lv == 3 then
		return 81
	elseif lv == 4 then
		return 95
	else
		return 111
	end
end
