-- SKC23J104
#include "txt_23104.txt"

function init_data()
	skillflag 		= "23104"
	TXT_SkillName()
	boost 			= "EffAP"
	other_boost 	= "EffDP"
end

function start()
	init_data()
	consume 		= "MP"

	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 23104
	
	defender 		= GetCallerPtr()
	
	--attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect  = GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 11
	elseif lv == 3 then
		return 14
	elseif lv == 4 then
		return 18
	else
		return 21
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 55
	elseif lv == 2 then
		return 72
	elseif lv == 3 then
		return 90
	elseif lv == 4 then
		return 112
	else
		return 135
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -4
	elseif lv == 2 then
		return -6
	elseif lv == 3 then
		return -7
	elseif lv == 4 then
		return -9
	else
		return -11
	end
end
