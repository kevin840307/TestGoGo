#include "txt_30105.txt"

function init_data()
	skillflag 		= "30105"
	TXT_SkillName()
	boost			= "EffMAP"
	other_boost 	= "EffAP"
end

function start()
	init_data()
	consume 		= "MP"
	
	varyDelta 		= 10

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 30105
	
	defender 		= GetCallerPtr()
	
	skillLv 		= GetSkillLv()
	effect			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect	= GetPara_Other(skillLv)

	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 11
	elseif lv == 3 then
		return 14
	elseif lv == 4 then
		return 18
	else
		return 21
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 28
	elseif lv == 2 then
		return 45
	elseif lv == 3 then
		return 68
	elseif lv == 4 then
		return 95
	else
		return 128
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -8
	elseif lv == 2 then
		return -11
	elseif lv == 3 then
		return -14
	elseif lv == 4 then
		return -18
	else
		return -21
	end
end
