function start()
	MinHit=2
	MinHitDelta=3
	BlockByShield = 1
	consume = "MP"
--	conTotal = 20
	APcount = "AP"
	DPcount = "DP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	
	conTotal = GetConTotal(skillLv)
		
	Attack()
end

function GetPara(lv)
	if lv == 1 then
		return 67
	elseif lv == 2 then
		return 85
	elseif lv == 3 then
		return 107
	elseif lv == 4 then
		return 130
	else
		return 157
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 72
	elseif lv == 2 then
		return 90
	elseif lv == 3 then
		return 112
	elseif lv == 4 then
		return 135
	else
		return 162
	end
end
