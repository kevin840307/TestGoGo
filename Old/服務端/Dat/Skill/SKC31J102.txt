#include "txt_31102.txt"

function start()
	SetSkillDefender(GetCallerPtr())
	Step01()
end

function Step01()
	Ask("n"..TXT_AskBasic(),TXT_Yes(),TXT_No(), "Choice_Step01")
end

function Choice_Step01()
	if GetChoice() == 0 then
		Step02()
	end
	QuestEnd()
end

function Step02()
	Ask("n"..TXT_Which(),TXT_Iron(),TXT_Crystal(),TXT_Grass(),TXT_Other(), "Choice_Step02")
end

function Choice_Step02()
	if GetChoice() == 0 then
		Step05()
	elseif GetChoice() == 1 then
		Step06()
	elseif GetChoice() == 2 then		
		Step07()
	else
		Step03()	
	end
end

function Step03()
	Ask("n"..TXT_Which(),TXT_Wood(),TXT_Fur(),TXT_Fruit(),TXT_Other(), "Choice_Step03")
end

function Choice_Step03()
	if GetChoice() == 0 then
		Step08()
	elseif GetChoice() == 1 then
		Step09()
	elseif GetChoice() == 2 then		
		Step10()
	else
		Step04()	
	end
end

function Step04()
	Ask("n"..TXT_Which(),TXT_Shell(),TXT_Feather(),TXT_Other(),TXT_Leave(), "Choice_Step04")
end

function Choice_Step04()
	if GetChoice() == 0 then
		Step11()
	elseif GetChoice() == 1 then
		Step12()
	elseif GetChoice() == 2 then		
		Step02()
	else
		QuestEnd()
	end
end

function Step05()
