#include "txt_22101.txt"

function init_data()
	skillflag	= "22101"
	TXT_SkillName()
	boost 		= "EffMAP"
end

function start()
	init_data()
	consume 	= "MP"
	varyDelta 	= 20

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 22101
	
	defender 	= GetCallerPtr()
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 1
	elseif lv == 2 then
		return 2
	elseif lv == 3 then
		return 4
	elseif lv == 4 then
		return 7
	else
		return 11
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 15
	elseif lv == 3 then
		return 28
	elseif lv == 4 then
		return 45
	else
		return 68
	end
end
