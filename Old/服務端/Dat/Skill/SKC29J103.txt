--SKC29J103
#include "txt_29103.txt"

function init_data()
	skillflag 	= "29103"
	TXT_SkillName()
	boost		= "EffSpeed"
end

function start()
	init_data()
	skillLv 	= GetSkillLv()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= GetDuration(skillLv)
	duration 	= recastTime
	
	skillid 	= 29103
	
	defender 	= GetCallerPtr()
	
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)

	GoBoost()	
end

function GetDuration(lv)
	if lv == 1 then
		return 300
	elseif lv == 2 then
		return 600
	else
		return 900
	end
end

function GetPara(lv)
	return 1
end

function GetConTotal(lv)
	if lv == 1 then
		return 22
	elseif lv == 2 then
		return 72
	else
		return 135
	end
end
