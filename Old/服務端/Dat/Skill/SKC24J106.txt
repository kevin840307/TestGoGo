#include "txt_24106.txt"

function init_data()
	skillflag 	= "24106"
	TXT_SkillName()
	boost 		= "EffDP"
end

function start()
	init_data()
	consume 	= "Will"
	varyDelta 	= 20

	recastTime 	= 900
	duration 	= 900
	
	skillid 	= 24106
	
	defender 	= GetCallerPtr()	
	
	attackType	= GetType()
	skillLv 	= GetSkillLv()
	effect 		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 6
	elseif lv == 2 then
		return 8
	elseif lv == 3 then
		return 12
	elseif lv == 4 then
		return 15
	else
		return 20
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 26
	elseif lv == 2 then
		return 35
	elseif lv == 3 then
		return 48
	elseif lv == 4 then
		return 59
	else
		return 74
	end
end
