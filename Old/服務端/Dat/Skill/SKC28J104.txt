#include "txt_28104.txt"

function start()
	consume 	= "MP"
	boost		= "HP"
	
	varyDelta 	= 10

	skillid 	= 28104
	TXT_SkillName()
	skillflag 	= "28104"
	
	defender	= GetTargetPtr()
	if defender == 0 then
		defender = GetCallerPtr()
	end
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	Cure_Range(400)
end

function GetPara(lv)
	if lv == 1 then
		return 28
	elseif lv == 2 then
		return 45
	elseif lv == 3 then
		return 68
	elseif lv == 4 then
		return 95
	else
		return 128
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 84
	elseif lv == 2 then
		return 135
	elseif lv == 3 then
		return 204
	elseif lv == 4 then
		return 285
	else
		return 384
	end
end
