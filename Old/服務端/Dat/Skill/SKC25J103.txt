--SKC25J103
#include "txt_25103.txt"

function init_data()
	skillflag 		= "25103"
	TXT_SkillName()
	boost 			= "EffHR"
	other_boost 	= "EffDR"
end

function start()
	init_data()
	consume 		= "MP"
	
	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 25103
	
	defender 		= GetCallerPtr()	
	
	--attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 5
	elseif lv == 2 then
		return 6
	elseif lv == 3 then
		return 7
	elseif lv == 4 then
		return 8
	else
		return 9
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 42
	elseif lv == 2 then
		return 55
	elseif lv == 3 then
		return 72
	elseif lv == 4 then
		return 90
	else
		return 112
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -5
	elseif lv == 2 then
		return -6
	elseif lv == 3 then
		return -7
	elseif lv == 4 then
		return -8
	else
		return -9
	end
end
