#include "txt_27107.txt"

function init_data()
	skillflag 	= "27107"
	TXT_SkillName()
	boost		= "EffDR"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 600
	duration 	= 600
	
	skillid 	= 27107
	
	defender	= GetTargetPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 7
	elseif lv == 2 then
		return 8
	elseif lv == 3 then
		return 9
	elseif lv == 4 then
		return 10
	else
		return 11
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 36
	elseif lv == 2 then
		return 45
	elseif lv == 3 then
		return 56
	elseif lv == 4 then
		return 68
	else
		return 81
	end
end
