#include "txt_28106.txt"

function init_data()
	skillflag 	= "28106"
	TXT_SkillName()
	boost		= "EffMPMAX"
end

function start()
	init_data()
	consume 	= "MP"
	
	varyDelta 	= 10

	recastTime 	= 600
	duration 	= 600
	
	skillid 	= 28106
	
	defender	= GetTargetPtr()
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	GoBoost()
end

function GetPara(lv)
	if lv == 1 then
		return 108
	elseif lv == 2 then
		return 134
	elseif lv == 3 then
		return 162
	elseif lv == 4 then
		return 194
	else
		return 228
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 45
	elseif lv == 2 then
		return 56
	elseif lv == 3 then
		return 68
	elseif lv == 4 then
		return 81
	else
		return 95
	end
end
