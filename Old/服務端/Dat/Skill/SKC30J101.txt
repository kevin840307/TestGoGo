#include "txt_30101.txt"

function start()
	consume 	= "MP"
	boost		= 281
	
	varyDelta 	= 10

	skillid 	= 30101
	TXT_SkillName()
	skillflag 	= "30101"
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	Acquest()
end

function GetPara(lv)
	if lv == 1 then
		return 100
	elseif lv == 2 then
		return 150
	elseif lv == 3 then
		return 200
	elseif lv == 4 then
		return 250
	else
		return 300
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 15
	elseif lv == 3 then
		return 28
	elseif lv == 4 then
		return 45
	else
		return 68
	end
end
