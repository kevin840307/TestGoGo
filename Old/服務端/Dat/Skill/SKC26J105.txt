--SKC26J105
#include "txt_26105.txt"

function init_data()
	skillflag 		= "26105"
	TXT_SkillName()
	boost 			= "EffMAP"
	other_boost 	= "EffDP"
end

function start()
	init_data()
	consume 		= "MP"

	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 26105
	
	defender 		= GetCallerPtr()	
	
	--attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 8
	elseif lv == 2 then
		return 11
	elseif lv == 3 then
		return 14
	elseif lv == 4 then
		return 18
	else
		return 21
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 28
	elseif lv == 2 then
		return 36
	elseif lv == 3 then
		return 45
	elseif lv == 4 then
		return 56
	else
		return 68
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -4
	elseif lv == 2 then
		return -6
	elseif lv == 3 then
		return -7
	elseif lv == 4 then
		return -9
	else
		return -11
	end
end
