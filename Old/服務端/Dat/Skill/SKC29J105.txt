function start()
	MinHit=2
	MinHitDelta=3
	BlockByShield = 1
	consume = "MP"
--	conTotal = 20
	APcount = "AP"
	DPcount = "DP"
	varyDelta = 20

	defender=GetTargetPtr()
	attackType=GetType()
	defenderType=GetTargetType(defender)
	skillinfo = GetSkillInfo()
	skillLv = GetTargetData(skillinfo, "Lv")
	effect = GetPara(skillLv)
	
	conTotal = GetConTotal(skillLv)
		
	Attack()
end

function GetPara(lv)
	if lv == 1 then
		return 37
	elseif lv == 2 then
		return 50
	elseif lv == 3 then
		return 67
	elseif lv == 4 then
		return 85
	else
		return 107
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 28
	elseif lv == 2 then
		return 37
	elseif lv == 3 then
		return 48
	elseif lv == 4 then
		return 60
	else
		return 75
	end
end
