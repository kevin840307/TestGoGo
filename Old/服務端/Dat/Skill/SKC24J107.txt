--SKC24J107
#include "txt_24107.txt"

function init_data()
	skillflag 		= "24107"
	TXT_SkillName()
	boost 			= "EffHPMAX"
	other_boost 	= "EffDP"
end

function start()
	init_data()
	consume 		= "MP"
	
	varyDelta 		= 20

	recastTime 		= 900
	duration 		= 900
	
	skillid 		= 24107
	
	defender 		= GetCallerPtr()	
	
	--attackType		= GetType()
	skillLv 		= GetSkillLv()
	effect 			= GetPara(skillLv)
	conTotal 		= GetConTotal(skillLv)
	other_effect 	= GetPara_Other(skillLv)
	
	GoBoost_2_Boost()
end

function GetPara(lv)
	if lv == 1 then
		return 340
	elseif lv == 2 then
		return 416
	elseif lv == 3 then
		return 500
	elseif lv == 4 then
		return 592
	else
		return 692
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 112
	elseif lv == 2 then
		return 135
	elseif lv == 3 then
		return 162
	elseif lv == 4 then
		return 190
	else
		return 222
	end
end

function GetPara_Other(lv)
	if lv == 1 then
		return -9
	elseif lv == 2 then
		return -11
	elseif lv == 3 then
		return -13
	elseif lv == 4 then
		return -15
	else
		return -18
	end
end
