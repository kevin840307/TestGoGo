#include "txt_28103.txt"

function start()
	consume 	= "Will"
	boost		= "MP"
	
	varyDelta 	= 10

	skillid 	= 28103
	TXT_SkillName()
	skillflag 	= "28103"
	
	defender	= GetTargetPtr()
	if defender == 0 then
		defender = GetCallerPtr()
	end
	
	skillLv 	= GetSkillLv()
	effect		= GetPara(skillLv)
	conTotal 	= GetConTotal(skillLv)
	
	Cure()
end

function GetPara(lv)
	if lv == 1 then
		return 17
	elseif lv == 2 then
		return 21
	elseif lv == 3 then
		return 31
	elseif lv == 4 then
		return 41
	else
		return 53
	end
end

function GetConTotal(lv)
	if lv == 1 then
		return 17
	elseif lv == 2 then
		return 21
	elseif lv == 3 then
		return 31
	elseif lv == 4 then
		return 41
	else
		return 53
	end
end
