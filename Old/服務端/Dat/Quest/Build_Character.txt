--建立測試用角色

function start()
	Say("p1建立測試用角色")
	AddItem(10677)
	Ask("p需要什麼呢?","物品。","經驗值。","怒。","Check")
end


function Check()
	choice = GetChoice()
	if choice == 0 then
		GetItem()
	elseif choice == 1 then
		GetExp()
	else
		GetAngry()
	end
end

function GetAngry()
	SetTargetData(GetSummonedHorse(GetCallerPtr()),"Angry",100000000)
end

function GetExp()
	Ask("p需要多少經驗值呢?","100萬。","1000萬。","寵物經驗值","機甲經驗值","CheckExp")
end

function CheckExp()
	choice = GetChoice()
	if choice == 0 then
		Lv1Exp()
	elseif choice == 1 then
		Lv10Exp()
	elseif choice == 2 then
		GetPetExp()
	else
		GetMSEXP()
	end
end

function GetMSEXP()
	SetTargetData(GetSummonedHorse(GetCallerPtr()),"Exp",1000000)
end

function Lv1Exp()
	ModifyData("ExpAcc",10000000)
	ModifyData("ExpUse",10000000)
end

function Lv10Exp()
	ModifyData("ExpAcc",100000000)
	ModifyData("ExpUse",100000000)
end

function GetPetExp()
	SetTargetData(GetSummonedPet(GetCallerPtr()),"Exp",100000000)
	SetTargetData(GetSummonedPet(GetCallerPtr()),"Exp",100000000)
end
