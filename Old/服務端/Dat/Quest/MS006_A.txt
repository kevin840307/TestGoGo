--皇宮侍衛(1) (MS006_A)

--MS006_Task
-- 1: title shown
-- 2: quest started
-- 3: quest completed

--MS006_LostItem
-- 1: lose item 10299
-- 2: lose item 10300

--MS006_Done
-- 1: quest completed

--MS007_Task
-- 1: quest started
-- 2: go to find lolosie
-- 3: get the flower, go to find prince
-- 4: found prince, got the water
-- 5: get the report
-- 6: give the report to prince

function start()
	if GetFlag("MS005_Done")~=1 then
		dialogue_01()
	elseif GetFlag("MS006_Done")~=1 then
		status = GetFlag("MS006_Task")
		if status >=3 then
			dialogue_07()
		elseif status >=2 then
			go_check_status_2()
		else
			dialogue_02()
		end
	else
		status = GetFlag("MS007_Task")
		if status>=6 then
			dialogue_7_04()
		elseif status>=5 and GetCountItemCount(10331)>=1 then
			dialogue_7_03()
		elseif status>=1 then
			dialogue_7_02()
		else
			dialogue_7_01()
		end
	end
end

function dialogue_01()
	Say("n1大皇子偉爾他為人溫文、善良，未來皇國由他管理，大家也很放心。","End")
end

--玩家已完成MS005
function dialogue_02()
	Say("n1看你的衣著，你就是騎士普頓所說的熱心軍人嗎？")
	Ask("n普頓在大皇子面前提及過你，現在正是皇子用人之際，他對你的事十分感興趣，你願意去參見王子嗎？","這個當然願意。","對不起，恐怕現在不行。","reply_02")
end

function reply_02()
	choice = GetChoice()
	if choice == 0 then
		dialogue_03()
	else
		dialogue_04()
	end
end

--玩家完成MS006_A－Step02_Choice1
function dialogue_03()
	Say("n1現在我馬上帶你去參見大皇子，請你跟我來。","action_03")
end
function action_03()
	MovePlayer(10944,7136,16)
	QuestEnd()
end

--玩家完成MS006_A－Step02_Choice2
function dialogue_04()
	Say("n1你．．．居然王子的邀請你也拒絕．．．","End")
end


function go_check_status_2()
	if GetCountItemCount(10301)>=1 and GetCountItemCount(10302)>=1 then
		dialogue_06()
	else
		letter_1 = GetCountItemCount(10299)
		letter_2 = GetCountItemCount(10300)
		if letter_1<=0 and letter_2<=0 then
			dialogue_08()
		elseif letter_1<=0 then
			dialogue_08a()
		elseif letter_2<=0 then
			dialogue_08b()
		else
			dialogue_05()
		end
	end
end

--玩家完成MS006_C－Step02

end

--玩家完成MS006_A－Step07-Choice1
function dialogue_09()
	Ask("n你．．．！這樣吧，你給我8000星幣，我便硬著頭皮向王子請求再寫一封信給你吧。","好吧．．．請你幫忙了。","你是在搶劫嗎？我才不會給你。","reply_09")
end

function reply_09()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Money")>=8000 then
			dialogue_10()
		else
			Say("n1唔...似乎你沒有8000元，我不能幫你了。")
		end
	end
	QuestEnd()	
end

function dialogue_10()
	if ItemListCountLeft(0)<1 then
		Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		Say("p7失去8000元")
		ModifyData("Money", -8000)

		Say("n1你先等一會，我馬上侯回來。")
		Say("n7皇宮侍衛匆忙地走進皇宮")
		Say("n7．．．．．．")
		Say("n7．．．．．．")
		Say("n7皇宮侍衛匆忙地從皇宮出來")
		Say("n1幸好大皇子十分大方，這是你要的信件。","dialogue_12")
	end
end


function dialogue_12()
	if GetFlag("MS006_LostItem") == 1 then
		Say("p7得到“給古加尼的信”")
		AddItem(10299)
	else
		Say("p7得到“給古洛尼的信”")
		AddItem(10300)
	end
	QuestEnd()
end
