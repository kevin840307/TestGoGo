--Learn_Skill_2104(沙甲)

function start()
	Giveback()
	if GetData("Career") == 2 then
	CheckOther()
	else
	Say("p7只有守護劍士才可學習沙甲。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習沙甲啊。","End")
	elseif HasSkill(2104)==1 then
	Say("p7你已經學會了沙甲。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了土神的形象，這時祂對你說．．．")
		Ask("p你要學習沙甲嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將沙甲傳授給你．．．")
		Say("p7你已經學會了沙甲。","End")
		RemoveItem(10341)
		AddSkill(2104)
	else
	Say("p7需要學習沙甲時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10341)
end
