--Learn_Skill_5104(幽暗魔力)

function start()
	Giveback()
	if GetData("Career") == 5 then
	CheckOther()
	else
	Say("p7只有魔動使才可學習幽暗魔力。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習幽暗魔力啊。","End")
	elseif HasSkill(5104)==1 then
	Say("p7你已經學會了幽暗魔力。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了間黑魔獸的形象，這時祂對你說．．．")
		Ask("p你要學習幽暗魔力嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將幽暗魔力傳授給你．．．")
		Say("p7你已經學會了幽暗魔力。","End")
		RemoveItem(10344)
		AddSkill(5104)
	else
	Say("p7需要學習幽暗魔力時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10344)
end
