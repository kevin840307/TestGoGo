--Learn_Skill_2103(黑石破劍)

function start()
	Giveback()
	if GetData("Career") == 2 then
	CheckOther()
	else
	Say("p7只有守護劍士才可學習黑石破劍。","End")
	end
end

function CheckOther()
	if GetData("Level") < 20 then
	Say("p7你的等級仍未可學習黑石破劍啊。","End")
	elseif HasSkill(2103)==1 then
	Say("p7你已經學會了黑石破劍。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了土神的形象，這時祂對你說．．．")
		Ask("p你要學習黑石破劍嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將黑石破劍傳授給你．．．")
		Say("p7你已經學會了黑石破劍。","End")
		RemoveItem(10333)
		AddSkill(2103)
	else
	Say("p7需要學習黑石破劍時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10333)
end
