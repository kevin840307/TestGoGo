--NQ026_S6 綠魔空間入口

function start()
    if CheckItem(10548) ~= 1 then
        Say("p7沒有合適的鎖匙，不能進入。")
    else
        Ask("p要使用綠魔空間之匙進入綠魔空間嗎?","是的","不是","ans_handler")
    end    
end 

function ans_handler()
    if GetChoice()==0 then
        RemoveItem(10548)
        MovePlayer(10432,14368,59)
    else
        QuestEnd()
    end
end
