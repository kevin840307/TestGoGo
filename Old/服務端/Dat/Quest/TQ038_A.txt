function start()
	if GetFlag("TQ038_Done") > 0 then
		Step26()
	else
		if GetData("Level") < 30 then
			Step01()
		else
			if CheckItem(6905) ~= 1 then
				if ItemListCountLeft(4) < 1 then
					Step27()
				else
					Step02()
				end
			else
				if GetCountItemCount(6905) < 80 then
					Step02()					
				else
					Step27()
				end
			end
		end
	end
end
