--依加魯穩定器(MS004_E)

--MS004_Task
-- 1: quest start, go to 泰雲
-- 2: go to 依加魯穩定器
-- 3: go to 迪爾
-- 4: go to 資訊大樓
-- 5: go to 嘉迪尼事務長
-- 6: go to 格靈森林
-- 7: got the 野花電子卡
-- 8: 基地destoryed
-- 9: quest completed

function start()
	status = GetFlag("MS004_Task")
	if status >= 3 then
		dialogue_03()
	elseif status >= 2 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function dialogue_01()
	Say("p1依加魯穩定器，是用作將高純度的依加魯能量穩定成結晶狀的機器。","End")
end

--玩家完成了MS004D－Step02
function dialogue_02()
	Say("p1好，讓我收集一些依加魯能量液回去再詳加調查。","dialogue_02b")
end

function dialogue_02b()
	if ItemListCountLeft(0)<1 then
		Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		AddItem(10272)
		Say("p7得到“依加魯能量液樣本”","End")
		SetFlag("MS004_Task", 3)
	end
end

--玩家完成了MS004E－Step02
function dialogue_03()
	Say("p1壞掉了的依加魯穩定器","End")
end
