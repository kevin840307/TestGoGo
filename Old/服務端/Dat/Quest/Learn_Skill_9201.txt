--Learn_Skill_9201(煉藥術)

function start()
	Giveback()
	if GetData("Career") == 9 then
	CheckOther()
	else
	Say("p7只有道具使才可學習煉藥術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 40 then
	Say("p7你的等級仍未可學習煉藥術啊。","End")
	elseif HasSkill(9201)==1 then
	Say("p7你已經學會了煉藥術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了創造之神的形象，這時祂對你說．．．")
		Ask("p你要學習煉藥術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將煉藥術傳授給你．．．")
		Say("p7你已經學會了煉藥術。","End")
		RemoveItem(10348)
		AddSkill(9201)
	else
	Say("p7需要學習煉藥術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10348)
end
