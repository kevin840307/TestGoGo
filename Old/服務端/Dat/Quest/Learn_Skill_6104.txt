--Learn_Skill_6104(安寧符文)

function start()
	Giveback()
	if GetData("Career") == 6 then
	CheckOther()
	else
	Say("p7只有守護使才可學習安寧符文。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習安寧符文啊。","End")
	elseif HasSkill(6104)==1 then
	Say("p7你已經學會了安寧符文。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了日冕聖獸的形象，這時祂對你說．．．")
		Ask("p你要學習安寧符文嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將安寧符文傳授給你．．．")
		Say("p7你已經學會了安寧符文。","End")
		RemoveItem(10345)
		AddSkill(6104)
	else
	Say("p7需要學習安寧符文時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10345)
end
