function response1()

	if GetChoice()==0 then
		if GetData("Money")>=1000 then

			ModifyData("Money",-1000)
			speech2()
		else
			speech01()
		end

	else
		speech1()
	
	end



end

function LearningSkill(career)
	
	SetFlag("NQ011_Task",2)	
	Say("n1嗯～就是這個。")
	Say("n1你先把指環都套到手上。")
	RemoveItem(10171,3)
	Say("p1嗯？")
	race=GetData("Race")
	if race==2 then
		Ask("n你想要保護別人的能力或是如風一般的速度？","保護別人的能力。","如風一般的速度。","choosingskill")

	else
		skillID=crr*1000 + 102
		LearningSpeech()
	end

end

function choosingskill()
	if GetChoice()==0 then
		if GetData("Career")==3 then
			skillID=3105
		else
			skillID=7105
		
		end

	else
		if GetData("Career")==3 then
			skillID=3102
		else
			skillID=7102
		end
	end
	LearningSpeech()
end

function GetSkillName(ID)
	if ID==1102 then
		return "赤之星"

	elseif ID==2102 then
		return "鐵之星"

	elseif ID==3102 then
		return "風之紗"
	
	elseif ID==4102 then
		return "霧之影"

	elseif ID==5102 then
		return "暗之力"

	elseif ID==6102 then
		return "光之盾"

	elseif ID==7102 then
		return "風影咒"

	elseif ID==8102 then
		return "幻靈咒"

	elseif ID==3105 then
		return "風之輕撫"
	
	elseif ID==7105 then
		return "靈之輕撫"

	end
end
