function start()
    if GetFlag("NQ026_KILL_SEQ") == -1 then
        Say("p7體內出現了一道紫色的微弱魔獸能量。")
        SetFlag("NQ026_KILL_SEQ",4)
    else        
        Say("p7體內的魔獸能量消失了！")
        Say("p7感應到另一道魔獸能量出現。")
        Say("p7體內得到一道紫色的微弱魔獸能量。")
        SetFlag("NQ026_KILL_SEQ",4)
    end        
end
