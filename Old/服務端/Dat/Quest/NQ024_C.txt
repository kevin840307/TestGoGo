--皇城鐵道車長(NQ024_C)

--NQ024_Type
--0: 一般單程票。
--1: 單程優惠票。
--2: 皇城鐵道套票。

function get_time()
	now = GetRealTime()
	return (floor(now/60) - floor(now/600)*10)
end

function start()
	time = get_time()
	if time>4 then
		time = time-5
	end
	if time<2 or GetFlag("NQ024_DEBUG")==1 then
		dia_02()
	else
		dia_01(5-time)
	end
end

--<未到上車時間>
function dia_01(time)
	Say("s前往尼尼亞皇城的列車還有約"..time.."分鐘才到達本站。","End")
end

--<到了上車時間>
function dia_02()
	Ask("s前往尼尼亞皇城的列車已到達本站，你要乘搭嗎？","乘搭行駛「魔法之都幹線」往尼尼亞皇城的列車。","對不起，我要考慮一會。","reply_02")
end
function reply_02()
	if GetChoice()==0 then
		dia_03()
	end
	QuestEnd()
end

function dia_03()
	Ask("s請問你要使用什麼車票呢？","一般單程票。","單程優惠票。","皇城鐵道套票。","對不起，我要考慮一會。","reply_03")
end
function reply_03()
	ans=GetChoice()
	if ans>=3 then
		QuestEnd()
	else
		dia_04(ans)
	end
end

function dia_04(ticket)
	--一般單程票
	if ticket==0 then
		if GetCountItemCount(10505) <1 then
			no_ticket()
		else
			dia_04a(ticket)
		end
	--單程優惠票
	elseif ticket==1 then
		if GetCountItemCount(10508) <1 then
			no_ticket()
		elseif GetData("Level")>=25 then
			high_level()
		else
			dia_04a(ticket)
		end
	--皇城鐵道套票
	else
		if GetCountItemCount(10511) <1 then
			no_ticket()
		else
			dia_04a(ticket)
		end
	end
end

function dia_04a(ticket)
	SetFlag("NQ024_Type", ticket)
	Say("s嗯，你可以上車了。","dia_04a1")
end
function dia_04a1()
	Say("m#X_mas_card\\BG_STATION_01.spr#na#SE0136.mp3#","dia_04a6")
end
function dia_04a2()
	Say("m#X_mas_card\\BG_STATION_02.spr#na#SE0135.mp3#","dia_04a5")
end
function dia_04a3()
	Say("m#X_mas_card\\BG_STATION_03.spr#na#SE0134.mp3#","dia_04a4")
end
function dia_04a4()
	Say("m#X_mas_card\\BG_STATION_04.spr#na#SE0137.mp3#","dia_04a5")
end
function dia_04a5()
	Say("m#X_mas_card\\BG_STATION_05_KING.spr#na##")
	QuestEnd()
end
function dia_04a6()
	ticket = GetFlag("NQ024_Type")
	if ticket==0 then
		RemoveItem(10505)
	elseif ticket==1 then
		RemoveItem(10508)
	else
		RemoveItem(10511)
	end
	MovePlayer(6490,5704,47)
	dia_04a5()
end

function no_ticket()
	Say("s喔！你身上沒有適當的車票，你必須往票務員那兒購買車票才能上車。","End")
end
function high_level()
	Say("s喔！等級在24以上是不能使用優惠票的，你必須往票務員那兒購買適當的車票才能上車。","End")
end
