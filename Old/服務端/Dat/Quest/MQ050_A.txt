--MQ050_A 中級武術導師歐麗

--MQ050_Task
-- 1: quest start, find 大牧師
-- 2: player get the 魔動爐
-- 3: player returned the 魔動爐

--MQ050_Done
-- 1: quest completed

--MQ051_TaskA
-- 1: quest start, find古米爾

--MQ051_TaskB
-- 1: quest start, find路特

--MQ050_Done
-- 1: quest completed

--MQ052_Task
-- 1: quest started
-- 2: go to find 巴基木
-- 3: got the 魔法火藥
-- 4: got the 魔風彈

--MQ052_Done
-- 1: quest completed

--MQ053_Task
-- 1: quest started
-- 2: go to find 巴基木
-- 3: got the 魔法火藥
-- 4: got the 魔風彈

--MQ053_Done
-- 1: quest completed

function start()
	career=GetData("Career")
	if GetData("Level")<30 then
		dialogue_below30()
	elseif career==0 then
		dialogue_below30()
	elseif career==1 then
		goCareer_01()
	elseif career==2 then
		goCareer_02()
	elseif career==3 then
		goCareer_03()
	elseif career==4 then
		goCareer_04()
	else
		--道具使
		dialogue_item_man()
	end
end

function dialogue_item_man()
	Say("n1你不需要學習武術的。","End")
end

function dialogue_below30()
	Say("n1歡迎你來到中級的課室，我是中級導師歐麗，請多多指教。","End")
end

-- 與靈鬥士的對話 start
function goCareer_01()
	status = GetFlag("MQ050_Task")
	if HasSkill(1103)==1 then
		dialogue_1_03()
	elseif status>=1 then
		dialogue_1_02()
	elseif HasSkill(1102)~=1 then
		dialogue_below30()
	else
		dialogue_1_01()
	end
end

function dialogue_1_01()
	Say("n1歡迎你來到中級的課室，我是中級導師歐麗，…")
	Ask("n初級的武術你已經學會了吧？你現在要學習新的武術嗎？","是的！","暫時沒需要。","reply_1_01")
end

function dialogue_1_01a()
	Say("n1你目前要學的是烈火爆劍，烈火爆劍的力量源自烈火力量，要弱化敵人就等於要溶化他的裝備，要引發這種力量，…")
	Say("n1就要找尼尼亞皇城大聖殿中大牧師借用魔動爐，這種東西可以煉出火依加魯結晶，只是集中依加魯力，要成烈火爆劍就不會很難。","End")
	SetFlag("MQ050_Task", 1)
end

function dialogue_1_01b()
	Say("n1這是什麼話？算了。","End")
end

function reply_1_01()
	if GetChoice()==0 then
		dialogue_1_01a()
	else
		dialogue_1_01b()
	end
end

function dialogue_1_02()
	Say("n1去尼尼亞皇城找"..Color(0,45,198).."大牧師"..Color(0,0,0).."吧！她應該會有"..Color(0,45,198).."魔動爐"..Color(0,0,0).."的。","End")
end

function dialogue_1_03()
	Say("n1太好了，你已經學會了，很少有你這樣有潛質的學生呢！")
	Say("n1目前我已經沒什麼可以教你了，不如你前往"..Color(0,45,198).."星之湖"..Color(0,0,0).."找一個退隱騎士吧！他是我的師傅，應該會有東西教你的。","End")
	SetFlag("MQ050_Done", 1)
end
-- 與靈鬥士的對話 stop

function goCareer_03()
	status = GetFlag("MQ052_Task")
	if HasSkill(3103)==1 then
		dialogue_3_04()
	elseif status>=4 then
		dialogue_3_05()
	elseif status>=1 then
		if GetCountItemCount(10232)>=1 and GetCountItemCount(10233)>=1 then
			dialogue_3_03()
		else
			dialogue_3_02()
		end
	elseif HasSkill(3102)~=1 then
		dialogue_below30()
	else
		dialogue_3_01()
	end
end
