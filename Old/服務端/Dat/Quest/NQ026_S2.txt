--NQ026_S2 青魔空間入口

function start()
    if CheckItem(10544) ~= 1 then
        Say("p7沒有合適的鎖匙，不能進入。")
    else
        Ask("p要使用青魔空間之匙進入青魔空間嗎?","是的","不是","ans_handler")
    end    
end 

function ans_handler()
    if GetChoice()==0 then
        RemoveItem(10544)
        MovePlayer(1792,8182,59)
    else
        QuestEnd()
    end
end
