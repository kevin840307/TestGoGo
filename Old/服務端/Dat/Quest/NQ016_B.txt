--NQ016_B 艾伯博士

---status (NQ016_Task)
-- 1: quest start, go to find 艾伯博士
-- 2: go items for 艾伯博士
-- 3: go to find 穆蘭卡博士
-- 4: quest completed

function start()
	status = GetFlag("NQ016_Task")
	if status == 2 then
		if ItemListCountLeft(0)<1 then
			Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
		elseif check_item()==1 then
			dialogue_04()
		else
			dialogue_03()
		end
	elseif status==1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function check_item()
	if GetCountItemCount(10050)>=50 and GetCountItemCount(10063)>=50 and GetCountItemCount(10049)>=50 then
		RemoveItem(10050,50)
		RemoveItem(10063,50)
		RemoveItem(10049,50)
		return 1
	else
		return 0
	end
end

function dialogue_01()
	Say("n1呵呵，研究也運行得如火如荼～","End")
end
