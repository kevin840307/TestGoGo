--古加尼校長( MS006_D)

function start()
	if GetCountItemCount(10299)>=1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function dialogue_01()
	Say("n1我們騎士著重的是榮譽，而榮譽是透過自己個人的行為來建立的，亦只有自己才能能判斷榮譽的真偽。","End")
end

--手持「給古加尼的信」
function dialogue_02()
	Say("n1這個學員，你這麼匆忙有什麼事了？")
	Say("p1大皇子叫我把這封信交到你的手上。")
	Say("n1啊！是嗎？馬上快給我看看。","dialogue_02b")
end
function dialogue_02b()
	Say("p7失去“給古加尼的信”")

	Say("n7古加尼校長馬上把信細讀")
	Say("n1喔！．．．唉，原來這樣。")
	Say("n1皇子居然這麼關心我們．．．")
	Say("p1校長，有什麼事嗎？")
	Say("n1我想你近日也會聽到一些留言，說我們的學生正準備私下和魔法學園的學生決鬥，想不到消息已傳到大皇子那兒了。")
	Say("n1近日我已不斷制止學生的過渡激烈行為，我實在不想我兩兄弟的分歧，使兩所學園產生流血事件。")
	Say("p1兩兄弟？")
	Say("n1這裡很多人也知道，魔法學園的校長－古洛尼是我的弟弟，我們為了證明自己的實力比對方強，分別行上不同的道路...")
	Say("n1卻想不到會影響到我們的學生，還使大皇子為我們憂心。")
	Say("n1這樣吧，請你盡快"..Color(0,45,198).."把此信交給大皇子"..Color(0,0,0).."，說明我會盡一切努力避免衝突。","dialogue_02c")
end
function dialogue_02c()
	Say("p7得到“古加尼給大皇子的信”","End")
	RemoveItem(10299)
	AddItem(10301)
end
