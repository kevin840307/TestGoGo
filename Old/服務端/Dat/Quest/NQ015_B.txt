--NQ015_B  阿爾瑪



--statue(NQ015_Task) =1: receive the mission
-- 		     =2: go to find  阿爾瑪
--		     =3: find 月霧石
--		     =4: mission is completed

function start()

	statue=GetFlag("NQ015_Task")
	done15=GetFlag("NQ015_Done")	

	if statue==4 then
		Say("n1	我正在造戒指…！凱兒正在等著我吧！","End")	

	elseif statue==3 then
		if GetCountItemCount(10212)>=1 then
			if done15==1 then
				Action1b()
			else
				Action1a()
			end
		else
			Say("n1總之，在流星之河有一種叫月出鳥的動物，牠們最喜歡月霧石的了！拜託你… ","End")
		end
	elseif statue==2 then
		if done15==1 then
			Speech1b()
		else
			Speech1a()
		end		
		SetFlag("NQ015_D",0)
		SetFlag("NQ015_Task",3)
	else
		Say("n1我很想念我的小孩子…我真的很想回去… ","End")

	end


end

function Timeout()

	Say("n1這個…真的是月霧石！可惜已溶掉了…","End")


end
function Action1a()
	hr=GetGameHours()
	if hr-GetFlag("NQ015_Bird")>6 then
		Say("n1這個…真的是月霧石！可惜已溶掉了…","End")
		RemoveItem(10212)
		SetFlag("NQ015_Get",0)

	else
		Say("n1這個…真的是月霧石！謝謝你！我這下可以打造戒指了……")
		Say("n1拿著這個！我一定要把它送給你！")
		if ItemListCountLeft(1)<1 then
			Say("n1對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
		else
			Say("n7得到"..Color(0,45,198).."幸福鳥之帽"..Color(0,0,0).."")
			RemoveItem(10212)
			AddItem(10213)
			Say("n1嗯…也送多一個稱號給你吧！")
			Say("n7得到稱號"..Color(0,45,198).."幸福使者"..Color(0,0,0).."")
			AddTitle(12)
			Say("n1不過你一定要守秘密！知道嗎？我想給她一個驚喜！","End")
			QuestGoFinish()
		end
	end
	
end

		Say("n1拿著這個！我一定要把它送給你！")

end	

function QuestGoFinish()
	SetFlag("NQ012_Task",0)
	SetFlag("NQ013_Task",0)
	SetFlag("NQ014_Task",0)
	SetFlag("NQ015_Done",1)
	SetFlag("NQ015_Task",4)
end
