--皇城鐵道車長(NQ024_B)

--NQ024_Type
--0: 一般單程票。
--1: 單程優惠票。
--2: 皇城鐵道套票。

function get_time()
	now = GetRealTime()
	return (floor(now/60) - floor(now/600)*10)
end

function start()
	time = get_time()
	if time>4 then
		time = time-5
	end
	if time<2 or GetFlag("NQ024_DEBUG")==1 then
		dia_02()
	else
		dia_01(5-time)
	end
end

--<未到上車時間>
function dia_01(time)
	Say("s往各城市的列車還有約"..time.."分鐘才到達本站。","End")
end

--<到了上車時間>
function dia_02()
	Ask("s前往各城市的列車已到達本站，你要乘搭往那裡的列車呢？","行駛「魔法之都幹線」往魔法之都的列車。","行駛「騎士之城幹線」往騎士之城的列車。","行駛「機械之城幹線」往機械之城的列車。","對不起，我要考慮一會。","reply_02")
end

function reply_02()
	ans=GetChoice()
	if ans==0 then
		dia_03(0)
	elseif ans==1 then
		dia_03(1)
	elseif ans==2 then
		dia_03(2)
	else
		QuestEnd()
	end
end

function dia_03(target)
	if target==0 then
		Ask("s請問你要使用什麼車票呢？","一般單程票。","單程優惠票。","皇城鐵道套票。","我想去別的地方。","reply_03a")
	elseif target==1 then
		Ask("s請問你要使用什麼車票呢？","一般單程票。","單程優惠票。","皇城鐵道套票。","我想去別的地方。","reply_03b")
	else
		Ask("s請問你要使用什麼車票呢？","一般單程票。","單程優惠票。","皇城鐵道套票。","我想去別的地方。","reply_03c")
	end
end
function reply_03a()
	ans=GetChoice()
	if ans>=3 then
		dia_02()
	else
		dia_04(0, ans)
	end
end
function reply_03b()
	ans=GetChoice()
	if ans>=3 then
		dia_02()
	else
		dia_04(1, ans)
	end
end
function reply_03c()
	ans=GetChoice()
	if ans>=3 then
		dia_02()
	else
		dia_04(2, ans)
	end
end


--<一般單程票>
function dia_04(target, ticket)
	--魔法之都幹線
	if target==0 then
		--一般單程票
		if ticket==0 then
			if GetCountItemCount(10505) <1 then
				no_ticket()
			else
				dia_04a(ticket)
			end
		--單程優惠票
		elseif ticket==1 then
			if GetCountItemCount(10508) <1 then
				no_ticket()
			elseif GetData("Level")>=25 then
				high_level()
			else
				dia_04a(ticket)
			end
		--皇城鐵道套票
		else
			if GetCountItemCount(10511) <1 then
				no_ticket()
			else
				dia_04a(ticket)
			end
		end
	--騎士之城幹線
	elseif target==1 then
		--一般單程票
		if ticket==0 then
			if GetCountItemCount(10507) <1 then
				no_ticket()
			else
				dia_04b(ticket)
			end
		--單程優惠票
		elseif ticket==1 then
			if GetCountItemCount(10510) <1 then
				no_ticket()
			elseif GetData("Level")>=25 then
				high_level()
			else
				dia_04b(ticket)
			end
		--皇城鐵道套票
		else
			if GetCountItemCount(10511) <1 then
				no_ticket()
			else
				dia_04b(ticket)
			end
		end
	--機械之城幹線
	else
		--一般單程票
		if ticket==0 then
			if GetCountItemCount(10506) <1 then
				no_ticket()
			else
				dia_04c(ticket)
			end
		--單程優惠票
		elseif ticket==1 then
			if GetCountItemCount(10509) <1 then
				no_ticket()
			elseif GetData("Level")>=25 then
				high_level()
			else
				dia_04c(ticket)
			end
		--皇城鐵道套票
		else
			if GetCountItemCount(10511) <1 then
				no_ticket()
			else
				dia_04c(ticket)
			end
		end
	end
end
