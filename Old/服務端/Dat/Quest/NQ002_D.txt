function start()
	choice=random(1000)
	if choice <=500 then
		Explose()
	else
		ID, Name, Type, Total = GetItemData(choice)
		if ItemListCountLeft(Type)<Total then
			Explose()
		else
			Say("p4啦啦啦～是什麼來的呢～？")
			Say("p1原來是"..Name.."……嗯！太好了！")
			Say("p7得到"..Name)
			AddItem(ID,Total)
		end
	end
	QuestEnd()
end

function Explose()
	Say("p4啦啦啦～是什麼來的呢～？")
	Say("p4這是什麼怪東西？")
	Say("p1好像是火藥！！")
	Say("p3糟了！要爆炸了！！")
	SetData("HP",1)
end
