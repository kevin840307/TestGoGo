--Learn_Skill_10101(治靈術)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習治靈術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 10 then
	Say("p7你的等級仍未可學習治靈術啊。","End")
	elseif HasSkill(10101)==1 then
	Say("p7你已經學會了治靈術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習治靈術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將治靈術傳授給你．．．")
		Say("p7你已經學會了治靈術。","End")
		RemoveItem(10352)
		AddSkill(10101)
	else
	Say("p7需要學習治靈術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10352)
end
