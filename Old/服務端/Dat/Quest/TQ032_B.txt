--4: get the 10637

function start()
	state = GetFlag("TQ032_Task")

	if state>=4 then
		dia_06()
	elseif state>=3 then
		dia_05()
	elseif state>=2 then
		if CheckItem(10636)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif state>=1 then
		dia_02()
	else
		dia_01()
	end
end
