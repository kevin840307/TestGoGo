--Learn_Skill_9204(裝備精煉)

function start()
	Giveback()
	if GetData("Career") == 9 then
	CheckOther()
	else
	Say("p7只有道具使才可學習裝備精煉。","End")
	end
end

function CheckOther()
	if GetData("Level") < 50 then
	Say("p7你的等級仍未可學習裝備精煉啊。","End")
	elseif HasSkill(9204)==1 then
	Say("p7你已經學會了裝備精煉。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了創造之神的形象，這時祂對你說．．．")
		Ask("p你要學習裝備精煉嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將裝備精煉傳授給你．．．")
		Say("p7你已經學會了裝備精煉。","End")
		RemoveItem(10351)
		AddSkill(9204)
	else
	Say("p7需要學習裝備精煉時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10351)
end
