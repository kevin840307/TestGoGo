--尤梨(MQ006_A)

--MQ005_Task
--1: quest started
--2: talked with student
--3: quest done

function start()
	if GetData("Career")==8 and GetData("Level")>=40 then
		status = GetFlag("MQ005_Task")
		if status>=3 then
			dialogue_05()
		elseif status>=2 then
			if GetCountItemCount(10026)>=1 then
				dialogue_04()
			else
				dialogue_03()
			end
		elseif status>=1 then
			dialogue_03()
		else
			dialogue_02()
		end
	else
		dialogue_01()
	end
end

--<玩家條件不附合時>
function dialogue_01()
	Say("p7你看到老師正在用布刷著她的銃，此時她看了你一下。")
	Say("n1嘿嘿…什麼了嗎？")
	Say("p1沒什麼，老師手上的是什麼呢？")
	Say("n1這是魔銃喔！樣子很帥吧？")
	Say("p7老師繼續刷著魔銃。","End")
end

--<條件附合>
function dialogue_02()
	Say("n7這是魔銃士老師，叫作尤梨。她的閃避能力可真是高得很呢！此時她正在刷著她那把魔銃。")
	Say("p1老師，你是魔銃士吧？")
	Say("n1嗯，是呢。我的職業可是很厲害的喔！我可以得到雷的力量，聖光雷獸也幫助我們呢。")
	Say("p7點點頭")
	Say("n1魔銃士很會閃避，而且攻擊的距離也很遠喔！只是有一點兒缺點，就是我們會消耗子彈……而且物理的防禦會比較差，")
	Say("n1雖然是這樣，我想我們的閃避那麼高，應該很少會被打中才對！而我們的力量，主要是依靠“雷之依加魯能量微粒”的。")
	Ask("n有什麼事了嗎？","我想成為魔銃士！","想來知道魔銃士到底是什麼而已。","reply_02")
end

function reply_02()
	if GetChoice()==0 then
		dialogue_02a()
	else
		dialogue_02b()
	end
end

--選項：「我想成為魔銃士！」
function dialogue_02a()
	Say("n1嗯，太好了！可是要成為魔銃士就要先得到"..Color(0,45,198).."靈金彈"..Color(0,0,0).."。")
	Say("p1嗯？靈金彈？什麼來的啊？")
	Say("n1嗯…我不可以說給你知道呢，所有人都應該自己去找，不過，應該會有學生會知道喔…","End")
	SetFlag("MQ005_Task", 1)
end

--<選：「想來知道魔銃士到底是什麼而已。」>
function dialogue_02b()
	Say("n1原來是這樣，你考慮一下當魔銃士吧。","End")
end

--<玩家接下此任務後>
function dialogue_03()
	Say("n1加油喔！靈金彈應該不會很難拿到才對。","End")
end

--<完成條件後>
function dialogue_04()
	Ask("n你確定要成為魔銃士了？","是的！","我要再考慮一下。","reply_04")
end

function reply_04()
	if GetChoice()==0 then
		dialogue_04a()
	else
		dialogue_04b()
	end
end

--<選 是的！>
function dialogue_04a()
	Say("p7把靈金彈交出去。")
	Say("n1看著吧。")
	Say("n7只見尤梨把靈金彈放在桌上，再加幾個動作，一陣奇怪的閃光就突然發出！此時眼前突然一亮，你感覺到被雷之依加魯能量微粒包圍著。")
	Say("n1你已經是魔銃士了！這把屠龍神銃就送給你吧，就當作是獎勵。")
	if ItemListCountLeft(1)<1 then
			Say("n1對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
	else
		Say("p7得850Exp")
		Say("p7得到「魔銃使」的稱號")
		Say("p7得到「屠龍神銃」","dialogue_04b2")
	end
end
function dialogue_04b2()
	RemoveItem(10026)
	SetFlag("MQ006_Task", 3)
	SetFlag("MQ006_Done", 1)
	gainexp(850)
	AddTitle(20)
	AddJob(2)
	AddItem(8350)
	QuestEnd()
end

--<選　我要再考慮一下。>
function dialogue_04b()
	Say("n1好好考慮清楚囉。","End")
end

--<任務完成後>
function dialogue_05()
	Say("n1快而準的魔法很重要，多點練習才可以得到成果！","End")
end

function gainexp(exp)
	ModifyData("ExpAcc",exp)
	ModifyData("ExpUse",exp)
end
