function response1()
	choice=GetChoice()
	if choice==0 then
		Say("n1很抱歉，由於就職必須“皇家就職師－古尼達博士”認可才能就職，但他去了搜集研究資料，所以暫時不能讓你就職啊。")
		Say("n1若你急著要就職，便必須要親自找他了。")
		Say("n1有人說古尼達博士沿尼尼亞皇城南下了，你可以往那兒找找。","End")
		SetFlag("K27001_Task",1)
		SetFlag("K27001_Start",1)
	else 
		Say("n1若你需要就職，歡迎你隨時回來。","End")
	end
end

function promotion()
	choice=GetChoice()
	race=GetData("Race")

	if choice==0 then
		PromoteToFighter(race)

	elseif choice==1 then
		PromoteToMagic(race)
		
	elseif choice==2 then
		--Ask("n你想就職的職業是？","道具使。","精靈使。","還是多考慮一會。","PromoteToOther")
		Ask("n你決定就職成為“見習道具使”嗎？","對，我願意就職為\“見習道具使\”。","還是多考慮一會。","ConfirmItem")
	else
		QuestEnd()
	end
end

function start() 
	lv=GetData("Level")
	statue=GetFlag("K27001_Task")
	if lv<10 then	
		Say("n1你好，我是專門處理就職的侍衛，當你到達等級十便可找我查詢就職事宜。","End")
	
	elseif GetFlag("K27001_END")==1 then
		Say("n1你已選擇了你今後的道路，但最終結果如何，還是要看你怎樣走下去。","End")
	
	elseif statue==4 then
		Ask("n你是來就職的嗎？","我想查詢一些職業的資料。","對啊，我是來就職的。","response2")

	elseif statue==3 then
		Say("p7向皇家侍衛交出認可書")
		Say("n1啊！你已經找到了古尼達博士嗎。現在開始你隨時也可以來就職了，你現在可先回去想清楚想當的職業。","End")
		RemoveItem(10035)
		SetFlag("K27001_Task",4)

	elseif statue==1 or statue==2 then
		Say("n1有人說古尼達博士沿尼尼亞皇城南下了，你可以往那兒找找。","End")
	
	else
		Ask("n等級十或以上便可以就職，你需要就職嗎？","我想馬上就職。","沒事，只是逛逛而已。","response1")
	end
end
