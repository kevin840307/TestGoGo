--士兵‧巴力
--(TQ033_B)

function start()
	state = GetFlag("TQ033_Task")

	if state>=4 then
		dia_06()
	elseif state>=3 then
		dia_05()
	elseif state>=2 then
		if CheckItem(10639)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif state>=1 then
		dia_02()
	else
		dia_01()
	end
end

function dia_01()
	Say("n1斯卡爾將軍好像是中邪了似的，有什麼人可以幫我啊！？","End")
end
