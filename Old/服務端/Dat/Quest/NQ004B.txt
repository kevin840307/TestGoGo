-- NQ004_B 藥師柯玲

function start()

	statue=GetFlag("NQ_004_GotQuest")

	if statue == 1 then
		if CheckItem(10018)>0 then
			Say("n3謝謝惠顧！", "End")
		else
			Ask("n有骨傷藥賣哦～～～小伙子，你要一個嗎？一個才500元而已～！", "OK～這是500元～麻煩給我一個！", "不用了！", "Step2")
		end

	else
		Say("n1有骨傷藥賣哦～～～小伙子，你要一個嗎？一個才500元而已～！")
		Say("p1不用了！")
		Say("n1好吧……", "End")
	end
end


function Step2()
	if GetChoice() == 0 then
		if GetData("Money") >= 500 then
			Say("n1謝謝惠顧！")
			if ItemListCountLeft(0)<1 then
				Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
			else
				Say("老伯應該需要這個！快點拿回去給他才行。","End")
				ModifyData("Money", -500)
				AddItem(10018)
			end
		else
			Say("n1小伙子，你的錢不夠。", "End")
		end

	else
		Say("n1好吧……", "End")
	end

end
