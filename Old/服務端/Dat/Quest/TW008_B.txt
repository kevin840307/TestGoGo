function GetEmotionById(id)
	if id==13 then 
		return "揮手"
	elseif id==7 then 
		return "憤怒"
	elseif id==12 then 
		return "勝利"
	elseif id==6 then 
		return "頭暈"
	elseif id==11 then 
		return "示愛"
	elseif id==5 then 
		return "道歉"
	elseif id==10 then 
		return "高興"
	elseif id==4 then 
		return "睡覺"
	elseif id==9 then 
		return "跳舞"
	elseif id==3 then 
		return "跳土風舞"
	elseif id==8 then 
		return "哭泣"
	elseif id==2 then 
		return "擺攤子"
	else
		return " "
	end
end

function start()
	if GetFlag("TW008_A_Step") >= 2 then
		if GetFlag("TW008_B_Step") >= 2 then
			if GetFlag("TW008_B_Step") >= 4 then
				if GetFlag("TW008_B_Step") >= 7 then
					if GetFlag("TW008_B_Step") >= 9 then
						Step12()
					else
						Step09()
					end
				else
					Step06()
				end
			else
				Step04()
			end
		else
			Step02()
		end	
	else
		Step01()
	end
end

function Step06_Handler()
	if GetChoice() == 0 then
		Step07()
	elseif GetChoice() == 1 then
		Step08()	
	else
		QuestEnd()	
	end
end

function Step07()
	hasmember, teammember = GetTeamList_First()
	if hasmember ~= 1 then
		Step05()
	else	
		em = random(11) + 2
		Say("n1看好了～我在做什麼動作呢～")
		Say("n1．．．．．．")
		Say("n1．．．．．．")	
		Say("n1我在做。"..GetEmotionById(em))	
		Say("n1快叫你的朋友跟我做吧～")
		SetFlag("TW008_Emotion",em)
		SetFlag("TW008_B_Step",7)
	end
end

function Step09()

--＜玩家身上物品空間少於1＞→Step11
--＜玩家團隊中沒有人在做出指定的動作＞→Step10
	hasmember, teammember = GetTeamList_First()
	IsNotValidEmotion = true
	while hasmember == 1 do
		if GetTargetEmotion(teammember) == GetFlag("TW008_Emotion") then
			IsNotValidEmotion = false
		end
		hasmember, temmember = GetTeamList_Next()
	end
	if IsNotValidEmotion then
		Step10()
	else	
		if ItemListCountLeft(0) < 1 then
			Step11()
		else
			Say("n1呵呵～對了！對了！就是這樣！")
			Say("p7就這樣，跟小米古玩了幾個小時．．．")

			Say("p7．．．．．．")
			Say("n1謝謝你們陪我玩了這樣久～^_^")
			Say("n1今天真的很高興～")
			Say("p7這時你在小米古身上得到了一點善良。")
			AddItem(10704)
			SetFlag("TW008_B_Step",9)
		end
	end
end
