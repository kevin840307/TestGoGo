--NQ026_S7 靛魔空間入口

function start()
    if CheckItem(10549) ~= 1 then
        Say("p7沒有合適的鎖匙，不能進入。")
    else
        Ask("p要使用靛魔空間之匙進入靛魔空間嗎?","是的","不是","ans_handler")
    end    
end 

function ans_handler()
    if GetChoice()==0 then
        RemoveItem(10549)
        MovePlayer(20416,15776,59)
    else
        QuestEnd()
    end
end
