--TQ013_B YoYo郡主的侍女

--TQ013_Done
-- 1: got the crown

function start()
	if GetFlag("TQ013_Done")==1 then
		dialogue_03()
	elseif RemoveItem(10254)==1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function dialogue_01()
	Say("n1你沒有古尼結晶石，不可以回答我的問題呢？","End")
end

--玩家手持古尼結晶石
function dialogue_02()
	Ask("n請問你YoYo郡主她最喜歡吃什麼蛋糕？\n請不要胡亂作答，不然你要再找一個古尼結晶石回來。","芒果蛋糕","士多啤梨蛋糕","巧克力蛋糕","栗子蛋糕","reply_02")
end

function reply_02()
	choice = GetChoice()
	if choice == 0 then
		if ItemListCountLeft(0)<1 then
			Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
		else
			Say("n1答對了，這YoYo之冠是給你的。")
			Say("p7得到“YoYo之冠”","End")
			AddItem(10255)
			SetFlag("TQ013_Done", 1)
		end
	else
		Say("n1真可惜，你答錯了，請再找一個古尼結晶石回來吧。","End")
	end
end

function dialogue_03()
	Say("n1恭喜你！你已經答對了問題！","End")
end
