--TQ005A  道具合成大師‧史卡


function start()

	crr=GetData("Career")
	
	if crr~= 9 then
		Say("n1做合成的人，要有靈活的手。")
		Say("p1啊～是嗎？","End")

	elseif HasSkill(9101)==1 and HasSkill(9102)==1 and HasSkill(9103)==1 then
		Say("n1到你的手更靈活的時候，我才教你更複習的合成技能吧！","End")	

	elseif HasSkill(9104)==1 then
		Say("n1你現在已有足夠的能力去學習一些簡單的合成，只要你肯學，我就教你一些簡單的合成技能吧。","End")
		Ask("n你想學哪一個技能？","基礎製藥","強化劑合成","攻擊物品製造","等一下才學","LearningSkills")
	
	else

		Say("n1做合成的人，要有靈活的手。可惜你的手仍不夠靈活，你至少學會了一項合成技能之後再回來找我吧。","End")
		

	end


end


function LearningSkills()

	choice=GetChoice()
	
	if choice==3 then 
		Say("n1好吧, 想學的話隨時找我。","End")
	
	else


		if choice==0 then
			if HasSkill(9101)==1 then
				HadLearnt()
			else
				Learn1()
			end
	
		elseif choice==1 then
			if HasSkill(9102)==1 then
				HadLearnt()
			else
				Learn2()
			end

		else 
			if HasSkill(9103)==1 then
				HadLearnt()
			else
				Learn3()
			end

		end
	end

end

function HadLearnt()
	Say("n1這個我已經教過你的了。","End")

end
