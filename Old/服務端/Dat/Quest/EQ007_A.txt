function Step01()
	if GetData("Bonnet") == 10696 or GetData("Bonnet") == 10697 or GetData("Bonnet") == 10698 or GetData("Bonnet") == 10699 or GetData("Bonnet") == 10700 then
		Say("n1請不要裝備以下物品：世運第１環、世運第２環、世運第３環、世運第４環、世運第５環或星空世運盾牌。")
	elseif GetData("Shield") == 10731 then
		Say("n1請不要裝備以下物品：世運第１環、世運第２環、世運第３環、世運第４環、世運第５環或星空世運盾牌。")
	else
		Ask("n這是星空世運會的頒獎典禮，你想來換取什麼獎品呢？","冠軍獎","亞軍獎","季軍獎","殿軍獎","Step01_Handler")
--＜選項＞
--．冠軍獎→Step 02
--．亞軍獎→Step 03
--．季軍獎→Step 04
--．殿軍獎→Step 05
	end
end

function Step01_Handler()
	if GetChoice() ==0 then
		Step02()
	elseif GetChoice() ==1 then
		Step03()
	elseif GetChoice() ==2 then
		Step04()
	elseif GetChoice() ==3 then
		Step05()
	end
end

		Step06()
	else
--＜玩家身上機甲空間少於3＞→Step 14
--＜玩家沒有以下物品＞→Step 06
--(10696、10697、10698、10699、10700、10731)
		Say("n1冠軍獎是世運冠軍機甲一套。")
		Ask("n你要用世運第１環、世運第２環、世運第３環、世運第４環、世運第５環和星空世運盾牌去換取冠軍獎嗎？","是的","不是","Step02_Handler")
--＜選項＞
--．是的→Step10
--．不是→Quest End
	end
end
