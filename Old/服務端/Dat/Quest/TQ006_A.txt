function response1()
	
	if GetChoice()==0 then
		GiveMoonCake()
	else
		Say("n1啊！那我一定不能準備好足夠的月餅了，怎麼辦好…","End")
	end


end

function GiveMoonCake()

		Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")

	end
end
