--Snow_Left_Hand 冰雪手部裝備卡(10372)

function start()
	dialogue_01()
end

function dialogue_01()
	level = GetData("Level")
	item_id1, item_id1b, item_name1, item_id2, item_id2b, item_name2, desb_1, desb_2 = get_item_data(level)
	if level<10 then
		get_item(item_id1, item_id1b, item_name1)
	else
		Ask("p請選擇你想要的裝備：", desb_1..item_name1, desb_2..item_name2, "reply_01")
	end
	QuestEnd()
end

function reply_01()
	level = GetData("Level")
	item_id1, item_id1b, item_name1, item_id2, item_id2b, item_name2, desb_1, desb_2 = get_item_data(level)
	ans = GetChoice()
	if ans == 0 then
		get_item(item_id1, item_id1b, item_name1)
	else
		get_item(item_id2, item_id2b, item_name2)
	end
end

function get_item(id, id2, name)
	if ItemListCountLeft(1)<2 then
		Say("p7對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
		AddItem(10372)
	else
		AddItem(id)
		AddItem(id2)
		Say("p7得到"..name, "End")
	end
end
