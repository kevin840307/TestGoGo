--Learn_Skill_10114(操獸術)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習操獸術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 18 then
	Say("p7你的等級仍未可學習操獸術啊。","End")
	elseif HasSkill(10114)==1 then
	Say("p7你已經學會了操獸術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習操獸術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將操獸術傳授給你．．．")
		Say("p7你已經學會了操獸術。","End")
		RemoveItem(10365)
		AddSkill(10114)
	else
	Say("p7需要學習操獸術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10365)
end
