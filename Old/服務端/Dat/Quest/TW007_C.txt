function start()
	if ItemListCountLeft(0) < 1 then
		dia_02()
	elseif GetCountItemCount(71)>299 then
		dia_03()
	elseif GetCountItemCount(6909)>298 then
		dia_04()
	elseif GetCountItemCount(6905)>299 then
		dia_05()
	elseif GetCountItemCount(221)>299 then
		dia_06()

		dia_07()
	elseif GetCountItemCount(10238)>299 then
		dia_08()
	elseif ItemListCountLeft(1)<1 then
		dia_09()
	elseif GetCountItemCount(10047)>290 then
		dia_10()
	elseif GetCountItemCount(70)>297 then
		dia_11()
	else
		main()
	end
end
