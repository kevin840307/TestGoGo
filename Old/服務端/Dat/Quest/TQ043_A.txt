function start()
	if GetFlag("TQ041_Done") == 1 then
		if GetFlag("TQ043_A_Step")>=2 then
			if GetFlag("TQ043_A_Step") >= 3 then
				if GetFlag("TQ043_C_Step") >= 1 then
					if GetFlag("TQ043_A_Step") >= 7 then
						if GetFlag("TQ043_D_Step") >= 2 then
							if GetFlag("TQ043_C_Step") >= 3 then
								if GetFlag("TQ043_E_Step")>= 7 then
									if GetFlag("TQ043_A_Step")>= 11 then
										if GetFlag("TQ043_A_Step")>=13 then
											Step15()
										else
											if (GetRealTime() - GetFlag("TQ043_A_TC")) < 5400 then
												Step12()
											else
												Step13()
											end
										end
									else
										Step11()
									end
								else
									Step10()
								end
							else
								Step09()
							end
						else
							Step08()
						end
					else
						Step06()
					end
				else
					Step04()
				end
			else
				Step03()	
			end
		else
			Step02()		
		end
	else
		Step01()
	end
end

function Step01()
	Say("n1牠們的力量太大了，平凡人都難以承受。")
	SetFlag("TQ043_A_TC",0)
end

function Step02()
--＜玩家已經完成TQ041（首步曲‧不能打開的星空之門）＞
	Say("p1你怎麼會傷得這麼重啊？快找醫師。")
	Say("n1我的名字是豆貝斯，剛由天空島逃出來的人，本來，一共有四人一起和我到天空島，但現在．．．")
	Say("p1其他人怎麼了？天空島一直也很平靜的，怎會令到你們受傷。")
	Say("n1啊！我的同伴都已經離我而去了，嗚嗚。那些聖獸侵襲我們，就只有我一個人有幸逃出來。")
	Say("p1是聖獸？怎麼會．．．哎呀，他暈死去了，要快找道具使朋友買瓶超級治療藥水來救他。")
	SetFlag("TQ043_A_TC",0)
	SetFlag("TQ043_A_Step",2)
end	
function Step03()
--＜After TQ043_A．Step02＞
--＜玩家持有超級治療藥水(135)＞
--＜玩家身上沒有超級治療藥水(135)＞→Step05
	if CheckItem(135) ~= 1 then 
		Step05()
	else		
		Say("p1豆貝斯，先吃下這些藥吧。")
	--＜收去玩家一個超級治療藥水(135)＞
		Say("n6唔，這裡就是天堂嗎？我也死了吧。")
		Say("p3醒過來啊，豆貝斯，這裡還未是天堂，你還活著。")
		Say("p1請告訴我，發生了什麼事？天空島內的情況怎樣？事態很嚴重了吧。")
		Say("n6天空島內的聖獸突然襲擊人民，而且還有其他魔物出現，快．．．快通知皇城。")
		Say("p1不是吧，那麼嚴重．．．還是先向基斯將軍回報情況。")
		Say("p1豆貝斯，你先好好休息，事件我會替你回報，別擔心。")
		RemoveItem(135)
		SetFlag("TQ043_A_Step",3)
	end
end

--．其實我自己也怕，還是聽你的話。→Quest End
--．不可以，此事很嚴重，一定要調查清楚。→Step07
end 

function Step06_Handler()
	if GetChoice()==0 then
		QuestEnd()
	else
		SetFlag("TQ043_A_Step",6)
		Step07()
	end
end

function Step13()
--＜After TQ043_A．Step11＞
--＜計時後已過一小時三十分鐘＞
--＜玩家身上沒有足夠空間得到蝙蝠之吻(10748)＞→Step14
	if ItemListCountLeft(0) < 1 then
		Step14()
	else
		Say("p1怎樣，成功嗎？")
		Say("n1還好吧，你要的東西我給你取到了。")
		Say("p7得到『蝙蝠之吻』")
		Say("p1豆貝斯，你是怎樣取得這個『蝙蝠之吻』的？")
		Say("n1這個是秘密，我是不會告訴你的。")
		Say("p1．．．．．．")
		AddItem(10748)
		SetFlag("TQ043_A_Step",13)
	end
end
