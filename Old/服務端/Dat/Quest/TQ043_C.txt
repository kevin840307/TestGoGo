function start()
	if GetFlag("TQ043_B_Step") >= 3 then
		if GetFlag("TQ043_C_Step")>=1 then
			if GetFlag("TQ043_B_Step")>=5 then
				if GetFlag("TQ043_C_Step")>=3 then
					if GetFlag("TQ043_D_Step")>=6 then
						if GetFlag("TQ043_B_Step")>=6 then
							if GetFlag("TQ043_C_Step")>=6 then
								Step07()
							else
								Step06()
							end
						else
							Step05()
						end
					else
						Step04()
					end
				else
					Step03()
				end
			else
				Step02()
			end
		else	
			Step01()
		end
	else
		Step01()
	end
end
