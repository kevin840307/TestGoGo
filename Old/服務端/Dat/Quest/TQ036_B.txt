function start()
	if GetFlag("TQ036_Task") < 1 then
		if GetData("Level") > 29 then
			Step02()
		else
			Step01()
		end
	elseif GetFlag("TQ036_Task") == 2 then
		regetItem()
	else
		Step04()
	end
end

			itemSpace = 2
		end		
		if GetFlag("TQ036_Task") >= 2 then
			QuestEnd()
		else

				Step07()
			else
				Say("n1這些是你勝出比賽的獎勵！！")
