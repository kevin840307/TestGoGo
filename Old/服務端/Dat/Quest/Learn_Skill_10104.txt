--Learn_Skill_10104(治療神咒)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習治療神咒。","End")
	end
end

function CheckOther()
	if GetData("Level") < 50 then
	Say("p7你的等級仍未可學習治療神咒啊。","End")
	elseif HasSkill(10104)==1 then
	Say("p7你已經學會了治療神咒。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習治療神咒嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將治療神咒傳授給你．．．")
		Say("p7你已經學會了治療神咒。","End")
		RemoveItem(10355)
		AddSkill(10104)
	else
	Say("p7需要學習治療神咒時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10355)
end
