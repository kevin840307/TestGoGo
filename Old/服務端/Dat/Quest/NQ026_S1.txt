--NQ026_S1 赤魔空間入口

function start()
    if CheckItem(10543) ~= 1 then
        Say("p7沒有合適的鎖匙，不能進入。")
    else
        Ask("p要使用赤魔空間之匙進入赤魔空間嗎?","是的","不是","ans_handler")
    end    
end 

function ans_handler()
    if GetChoice()==0 then
        RemoveItem(10543)
        MovePlayer(4160,2656,59)
    else
        QuestEnd()
    end
end
