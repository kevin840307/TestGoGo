--Learn_Skill_7104(清風符文)

function start()
	Giveback()
	if GetData("Career") == 7 then
	CheckOther()
	else
	Say("p7只有聖靈使才可學習清風符文。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習清風符文啊。","End")
	elseif HasSkill(7104)==1 then
	Say("p7你已經學會了清風符文。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了暴風聖獸的形象，這時祂對你說．．．")
		Ask("p你要學習清風符文嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將清風符文傳授給你．．．")
		Say("p7你已經學會了清風符文。","End")
		RemoveItem(10346)
		AddSkill(7104)
	else
	Say("p7需要學習清風符文時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10346)
end
