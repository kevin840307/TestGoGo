--Learn_Skill_10102(治靈咒)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習治靈咒。","End")
	end
end

function CheckOther()
	if GetData("Level") < 15 then
	Say("p7你的等級仍未可學習治靈咒啊。","End")
	elseif HasSkill(10102)==1 then
	Say("p7你已經學會了治靈咒。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習治靈咒嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將治靈咒傳授給你．．．")
		Say("p7你已經學會了治靈咒。","End")
		RemoveItem(10353)
		AddSkill(10102)
	else
	Say("p7需要學習治靈咒時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10353)
end
