--堅定樹
--(TQ033_D)

#include "CommTeam.txt"

function start()
	if GetFlag("TQ033_C")<1 then
		dia_01()
	elseif CheckItem(10639)>0 then
		dia_06()
	else
		total = GetTeamTotal()
		if total >= 5 then
			if IsTeamCloseTo(15232,5760,43, 250) >0 then
				dia_04()
			else
				Say("p1這就是堅定樹吧！？")
				Say("p1隊員們，再走近一黠吧～","End")
			end
		elseif total >0 then
			dia_03()
		else
			dia_02()
		end
	end
end

function dia_01()
	Say("p7這就是堅定樹吧．．．","End")
end

--＜After TQ033_C Step02＞
--＜團隊中沒有人在指定範圍＞
function dia_02()
	Say("p1這就是堅定樹吧！？")
	Say("p1要找齊6個團員來到幫忙才可得到堅定樹的樹枝．．．","End")
end

--＜After TQ033_C Step02＞
--＜團隊中不足６人在指定範圍＞
function dia_03()
	Say("p1這就是堅定樹吧！？")
	Say("p3大家來出力折下樹枝吧！！")
	Say("p1．．．．．．")
	Say("p1．．．．．．")
	Say("p1不能折下樹技呢．．．果然是人數不足。","End")
end

--＜After TQ033_C Step02＞
--＜團隊中有6人在指定範圍＞
function dia_04()
	if ItemListCountLeft(0)<1 then
		dia_05()
	else
		Say("p1這就是堅定樹吧！？")
		Say("p3大家來出力折下樹枝吧！！")
		Say("p1．．．．．．")
		Say("p1．．．．．．")
		Say("p3啪！")
		Say("p3折下了樹枝！")
		Say("p7得到堅定樹的樹枝。","End")
		AddItem(10639)
		SetFlag("TQ033_C", 2)
	end
end

function dia_05()
	Say("p7身上沒有足夠的物品空間啊．．．先整理一下好了。","End")
end

--＜玩家身上持有堅定樹的樹枝(10639)＞
function dia_06()
	Say("p7不要過度採集啊！","End")
end
