function start()
	if GetFlag("TQ041_B_Step") >= 1 then
		if GetFlag("TQ041_B_Step") >=3 then
			if GetFlag("TQ041_I_Step") >=3 then
				if GetFlag("TQ041_H_Step") >=3 then
					if GetFlag("TQ041_I_Step") >= 5 then
						if GetFlag("TQ041_K_Step") >= 1 then
							Step07()
						else
							Step06()
						end
					else
						Step05()
					end
				else
					Step04()
				end
			else			
				Step03()
			end 
		else
			Step02()
		end
	else
		Step01()
	end
end
