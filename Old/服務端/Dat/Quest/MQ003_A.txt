--MQ003 魔動使就職
--謝利(MQ003_A)

--MQ003_Task
--1 Quest started
--2 Talked with student already
--3 Quest done

function start()
	status = GetFlag("MQ003_Task")
	if status>=3 then
		dialogue_06()
	elseif status>=1 then
		if GetCountItemCount(10006)>=1 and GetCountItemCount(10007)>=1 then
			dialogue_05()
		else
			dialogue_04()
		end
	elseif GetData("Career")==5 then
		if GetData("Level")>=40 then
			dialogue_03()
		else
			dialogue_01()
		end
	else
		dialogue_02()
	end
end

--＜玩家條件不附合時＞
function dialogue_01()
	Say("n1你想要研究黑暗的魔法嗎？呼呵呵呵。")
	Say("p7心驚...")
	Say("p1不、不用了。")
	Say("n1要40級以上的見習魔動使才可以成為這個職業呢，真的很可惜唷。")
	Say("n7老師又再做起自己的事來，不理你了。","End")
end

--<玩家不是機動靈時>(Deleted)
function dialogue_02()
	Say("n1嗯…你並不能成為魔動使唷…只有見習魔動使才可以成為這個職業呢，真的很可惜唷。","End")
end

--<條件附合>
function dialogue_03()
	Say("p1老師的工作是什麼樣的呢？")
	Say("n1老師是當魔動使的啊……這個工作可好玩的很呢，呵呵。魔動使可以操縱黑暗的力量，有著闇黑魔獸守護著，很酷吧對不？")
	Say("p7點點頭")
	Say("n1魔動使的魔法攻擊力可是很強的唷，沒有什麼可以抵抗我們呢！")
	Say("n1但是唯一的缺點，就是我們的物理防禦上比較弱，不過我們應該可以在別人衝過來前已經殺了他才對……嘿嘿嘿。")
	Say("n1順帶說明唷，我們魔法力的來源是大氣中的"..Color(0,45,198).."“黑暗依加魯能量微粒”"..Color(0,0,0).."的。")
	Ask("n說起來？為什麼突然問起我啦？呵呵。","我想成為魔動使！","沒什麼事，只是來看看老師。","reply_03")
end

function reply_03()
	if GetChoice()==0 then
		dialogue_03a()
	else
		dialogue_03b()
	end
end

--選項：「我想成為魔動使！」
function dialogue_03a()
	Say("n1呵呵……真是太好了，我很高興聽你這樣說呢……要成為魔動使，必須舉行一個儀式呢。而這個儀式嘛，需要有"..Color(0,45,198).."暗之光"..Color(0,0,0).."和"..Color(0,45,198).."魔物之血"..Color(0,0,0).."。")
	Say("p1嗯？這是什麼？")
	Say("n1自己努力去找吧，好好加油喔……不過呢，給你一個暗示吧，我想學校內一定會有學生知道的，你去問問學長們喔，呼呼……","End")
	SetFlag("MQ003_Task", 1)
end

--<選：「沒什麼事，只是來看看老師。」>

	Say("n1你有這個心真好……老師很高興喔。","End")
end

--<玩家接下此任務後>
function dialogue_04()
	Say("n1努力去找"..Color(0,45,198).."暗之光"..Color(0,0,0).."和"..Color(0,45,198).."魔物之血"..Color(0,0,0).."回來吧，好好加油喔……呼呵呵……","End")
end

--<完成條件後>
function dialogue_05()
	Ask("n你確定要成為魔動使了嗎……？呼呵呵…","是的！","我要再考慮一下。","reply_05")
end

function reply_05()
	if GetChoice()==0 then
		dialogue_05a()
	else
		dialogue_05b()
	end
end

function dialogue_05a()
	Say("n1嗯……很好。你看著唷。")
	Say("n7只見謝利把魔物之血滴了滴在暗之光上，本來平平無奇的石頭竟然在此時，發出奇異的光芒！")
	Say("n7你感覺到被黑暗依加魯能量微粒包圍著。")
	Say("n1嗯，好了，你已經是魔動使了唷……以後要加油喔！呼呵呵……這把屠龍魔杖，就送給你吧，當作是獎勵你的努力。")
	if ItemListCountLeft(1)<1 then
			Say("n1對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
	else
		Say("p7得到850Exp")
		Say("p7得到「魔動使」的稱號")
		Say("p7得到「屠龍魔杖」","dialogue_05a2")
	end
end
function dialogue_05a2()
	RemoveItem(10006)
	RemoveItem(10007)
	SetFlag("MQ003_Task", 3)
	SetFlag("MQ003_Done", 1)
	gainexp(850)
	AddTitle(17)
	AddJob(2)
	AddItem(8349)
	QuestEnd()
end

function dialogue_06()

end
