--Learn_Skill_10117(操獸咒)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習操獸咒。","End")
	end
end

function CheckOther()
	if GetData("Level") < 52 then
	Say("p7你的等級仍未可學習操獸咒啊。","End")
	elseif HasSkill(10117)==1 then
	Say("p7你已經學會了操獸咒。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習操獸咒嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將操獸咒傳授給你．．．")
		Say("p7你已經學會了操獸咒。","End")
		RemoveItem(10368)
		AddSkill(10117)
	else
	Say("p7需要學習操獸咒時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10368)
end
