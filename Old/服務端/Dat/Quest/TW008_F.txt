function start()
	if GetFlag("TW008_E_Step") >= 2 then
		if GetFlag("TW008_F_Step") >= 2 then
			Step03()
		else
			Step02()
		end
	else
		Step01()
	end		
end	
function Step01()
	Say("n1我老伴做的飯團是很美味的啊．．．")
end

	Say("p7得到美味飯糰收據。")

end
