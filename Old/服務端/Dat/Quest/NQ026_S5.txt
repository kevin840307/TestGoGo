--NQ026_S5 黃魔空間入口

function start()
    if CheckItem(10547) ~= 1 then
        Say("p7沒有合適的鎖匙，不能進入。")
    else
        Ask("p要使用黃魔空間之匙進入黃魔空間嗎?","是的","不是","ans_handler")
    end    
end 

function ans_handler()
    if GetChoice()==0 then
        RemoveItem(10547)
        MovePlayer(24448,3200,59)
    else
        QuestEnd()
    end
end
