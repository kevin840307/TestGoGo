--Learn_Skill_6103(魔光術)

function start()
	Giveback()
	if GetData("Career") == 6 then
	CheckOther()
	else
	Say("p7只有守護使才可學習魔光術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 20 then
	Say("p7你的等級仍未可學習魔光術啊。","End")
	elseif HasSkill(6103)==1 then
	Say("p7你已經學會了魔光術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了日冕聖獸的形象，這時祂對你說．．．")
		Ask("p你要學習魔光術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將魔光術傳授給你．．．")
		Say("p7你已經學會了魔光術。","End")
		RemoveItem(10337)
		AddSkill(6103)
	else
	Say("p7需要學習魔光術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10337)
end
