function start()
	Step01()
end

function Step01()
	if GetData("Bonnet") == 10696 or GetData("Bonnet") == 10697 or GetData("Bonnet") == 10698 or GetData("Bonnet") == 10699 or GetData("Bonnet") == 10700 then
		Say("n1請不要裝備以下物品：世運第１環、世運第２環、世運第３環、世運第４環、世運第５環。")
	else
		Ask("n這是星空世運會的頒獎典禮，你想來換取什麼獎品呢？","冠軍獎","亞軍獎","季軍獎","殿軍獎","Step01_Handler")
--＜選項＞
--．冠軍獎→Step 02
--．亞軍獎→Step 03
--．季軍獎→Step 04
--．殿軍獎→Step 05
	end
end

function Step01_Handler()
	if GetChoice() == 0 then
		Step02()
	elseif GetChoice() == 1 then
		Step03()
	elseif GetChoice() == 2 then
		Step04()
	elseif GetChoice() == 3 then	
		Step05()
	end
end

function Step02()
	if ItemListCountLeft(0) < 2 then
		Step14()
	elseif CheckItem(10696) ~= 1 or CheckItem(10697) ~= 1 or CheckItem(10698) ~= 1 or CheckItem(10699) ~= 1 or CheckItem(10700) ~= 1 then
		Step06()
	else
--＜玩家身上物品空間少於2＞→Step 14
--＜玩家沒有以下物品＞→Step 06
--(10696、10697、10698、10699、10700)
		Say("n1冠軍獎是天馬頭飾換領券和極級天馬裝備換領券。")
		Ask("n你要用世運第１環、世運第２環、世運第３環、世運第４環、世運第５環去換取冠軍獎嗎？","是的","不是","Step02_Handler")
--＜選項＞
--．是的→Step10
--．不是→Quest End
	end
end

function Step02_Handler()
	if GetChoice() == 0 then
		Step10()
	else
		QuestEnd()
	end
end

function Step03()
--＜玩家身上裝備空間少於1＞→Step 15
--＜玩家沒有以下物品＞→Step 07
--(10696、10697、10698、10699)
	if ItemListCountLeft(1) < 1 then
		Step15()
	elseif CheckItem(10696) ~= 1 or CheckItem(10697) ~= 1 or CheckItem(10698) ~= 1 or CheckItem(10699) ~= 1 then
		Step07()
	else
		Say("n1亞軍獎是白羊座頭飾。")
		Ask("n你要用世運第１環、世運第２環、世運第３環、世運第４環去換取亞軍獎嗎？","是的","不是","Step03_Handler")
--＜選項＞
--．是的→Step11
--．不是→Quest End
	end
end

function Step03_Handler()
	if GetChoice() == 0 then
		Step11()
	else
		QuestEnd()
	end
end

function Step04()
	if ItemListCountLeft(4) < 3 then
		Step16()
	elseif CheckItem(10696) ~= 1 or CheckItem(10697) ~= 1 or CheckItem(10698) ~= 1 then
		Step08()
	else
	
--＜玩家身上機甲裝備空間少於3＞→Step 16
--＜玩家身上機甲卡空間少於1＞→Step 16
--＜玩家沒有以下物品＞→Step 08
--(10696、10697、10698)
		Say("n1季軍獎是世運機甲卡。")
		Ask("n你要用世運第１環、世運第２環、世運第３環去換取季軍獎嗎？","是的","不是","Step04_Handler")
--＜選項＞
--．是的→Step12
--．不是→Quest End
	end
end

function Step04_Handler()

end

function Step05()
