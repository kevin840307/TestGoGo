--Snow_Sword 冰雪之劍裝備卡(10376)

function start()
	dialogue_01()
end

function dialogue_01()
	level = GetData("Level")
	item_id1, item_name1, item_id2, item_name2, desb_1, desb_2 = get_item_data(level)
	if level<10 then
		get_item(item_id1, item_name1)
	else
		Ask("p請選擇你想要的裝備：", desb_1..item_name1, desb_2..item_name2, "reply_01")
	end
	QuestEnd()
end

function reply_01()
	level = GetData("Level")
	item_id1, item_name1, item_id2, item_name2, desb_1, desb_2 = get_item_data(level)
	ans = GetChoice()
	if ans == 0 then
		get_item(item_id1, item_name1)
	else
		get_item(item_id2, item_name2)
	end
end

function get_item(id, name)
	if ItemListCountLeft(1)<1 then
		Say("p7對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
		AddItem(10376)
	else
		AddItem(id)
		Say("p7得到"..name, "End")
	end
end

function get_item_data(level)
	if level < 10 then
		return 20001, "「寒雪之劍」", 20001, "「寒雪之劍」", "等級10的裝備", "等級10的裝備"
	elseif level < 20 then
		return 20001, "「寒雪之劍」", 20002, "「雪崩之劍」", "等級10的裝備", "等級20的裝備"
	elseif level < 30 then
		return 20002, "「雪崩之劍」", 20003, "「雪龍之劍」", "等級20的裝備", "等級30的裝備"
	elseif level < 40 then
		return 20003, "「雪龍之劍」", 20004, "「雪戰龍之劍」", "等級30的裝備", "等級40的裝備"
	elseif level < 50 then
		return 20004, "「雪戰龍之劍」", 20005, "「雪龍王之劍」", "等級40的裝備", "等級50的裝備"
	elseif level < 60 then
		return 20005, "「雪龍王之劍」", 20006, "「雪戰神之劍」", "等級50的裝備", "等級60的裝備"
	elseif level < 70 then
		return 20006, "「雪戰神之劍」", 20007, "「寒雪戰神之劍」", "等級60的裝備", "等級70的裝備"
	else
		return 20007, "「寒雪戰神之劍」", 20008, "「雪龍神之劍」", "等級70的裝備", "等級80的裝備"
	end
end
