function start()
	AddItem(10458)
	go()
end
function go()
	Ask("p歡迎使用星空新手指南」～新手指導員","查看基本操作。","查看戰鬥資料。","查看城市資料。","還是不看了。","reply")
end

function reply()
	ans = GetChoice()
	if ans==0 then
		dia_01()
	elseif ans==1 then
		dia_02()
	elseif ans==2 then
		dia_03()
	else
		QuestEnd()
	end
end
