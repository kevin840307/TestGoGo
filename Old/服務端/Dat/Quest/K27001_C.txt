--K27001_C  星果立普

function start()
	if CheckItem(10036)>0 or GetFlag("K27001_Start")~=1 then
		QuestEnd()
	else
		if ItemListCountLeft(0)<1 then
			Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
		else
			Say("p7得到\“星果立普之羽\”","End")
			AddItem(10036,1)
			if CheckItem(10037)>0 and CheckItem(10038)>0 then
				SetFlag("K27001_GetAll",1)
			end
		end
	end		
	
end
