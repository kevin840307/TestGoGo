function start()
	if GetFlag("TQ043_A_Step") >= 7 then
		if GetFlag("TQ043_D_Step") >= 2 then
			if GetFlag("TQ043_C_Step")>=3 then
				if GetFlag("TQ043_D_Step") >= 6 then
					if GetFlag("TQ043_E_Step") >= 9 then
						if GetFlag("TQ043_D_Step") >= 9 then
							Step12()
						else
							Step09()
						end
					else
						Step08()
					end
				else
					Step05()
				end
			else
				Step04()
			end
		else
			Step02()
		end
	else
		Step01()
	end
end
