--Shop12 坐騎主體

function start()
	dialogue_Welcome()
end

function dialogue_Welcome()
	Ask("n歡迎光臨，你的坐騎需要維修嗎？","要","不要","reply_Welcome")
end

function reply_Welcome()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Career")==9 then
			OpenHorseEquipRepair (80)
		else
			OpenHorseEquipRepair (100)
		end
	else
		Say("n1歡迎下次再光臨。","End")
	end
end
