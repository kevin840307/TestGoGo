--星空詩人（NQ020_D）

function start()
	dia_01()
end

function dia_01()
	Ask("n你想聽一聽自古流傳的傳說嗎？","我想聽創世的故事。","我想聽神之戰爭的故事。","我想聽星之歷史。","稍後再聽吧。","reply_01")
end
function reply_01()
	ans = GetChoice()
	if ans==0 then
		dia_02()
	elseif ans==1 then
		dia_03()
	elseif ans==2 then
		dia_04()
	else
		QuestEnd()
	end
end

function dia_02()
	Say("n1一開始，世上沒有天，沒有地，只有一片混沌，一片迷霧，經過漫長的歲月。後來，迷霧一點一滴的積聚，讓世上出現了海洋─生命之海。")
	Say("n1而創世之神奧米加，就是生命之海所誕生的第一個生命。")
	Say("n1奧米加將迷霧造成天空，身體變成了大地，靈魂化做火紅的太陽，為大地帶來溫暖。奧米加就是這樣創造了艾達這個世界。艾達就由祂的三個兒女管理。")
	Say("n1祂們就是長子卡奧斯、次女佳雅和么子亞魯斯。")
	Say("n1三人合力共同管理艾達，像卡奧斯為世界創造了海中的生物，佳雅創造了地上的生物，而亞魯斯就創造了天空的生物。")
	Ask("n這就是這個世界的開始……","繼續看下一段。","稍後再看吧。","reply_02")
end
function reply_02()
	if GetChoice()==0 then
		dia_03()
	else
		QuestEnd()
	end
end

function dia_03()
	Say("n1本來三人安好地管理著艾達。可是，亞魯斯愛上了佳雅，二人因愛而生下了後代，那就是「人類」。")
	Say("n1因為人類擁有智慧，加上受到佳雅的教導，所以發展得非常快速與繁盛。卡奧斯覺得自己受到威脅，於是不斷為大地上帶來災難，同時更要求佳雅和自己繁衍後代。")
	Say("n1佳雅不願意，卡奧斯便開始了滅絕人類的行動！")
	Say("n1亞魯斯和佳雅為保護地上的人類，而與卡奧斯展開了神之戰爭！")
	Say("n1在神之戰爭裡，卡奧斯用力量迷惑不少人類成為了祂的戰士；亞魯斯因為被人類欺騙，因此遭到卡奧斯砍下一條手臂，而米古人就是在這個時候，從亞魯斯的血中誕生的。")
	Say("n1不過，也是從這個時候開始，亞魯斯就不再相信任何人類，只剩下佳雅與卡奧斯作戰。")
	Say("n1為了中止這場戰爭，佳雅犧牲了自己，將艾達分為三個星球，從此三人分開管治自己的星球。")
	Say("n1卡奧斯和受其迷惑的人類在古羅斯(Golues)建立了亞斯民俗，因為卡奧斯是挑起戰爭的源頭，所以在古羅斯不被奧米加祝福，導致當地氣候長期寒冷，沒有陽光。")
	Say("n1人類和米古人在佳雅所守護的加米蘭特星(Kamuland)定居下來，在這裡他們受到由神所製造的「機動靈」，幫助他們重建新故鄉。")
	Say("n1加上因為得到奧米加的祝福，加米蘭特成為氣候溫和，四季如春的美麗星球。")
	Say("n1不再相信人類的亞魯斯，獨自在尼路亞(Lalura)重新創造了各種野生動物，讓牠們在尼路亞自由地過著弱肉強食的生活。")
	Ask("n奧米加使三個星球每1000年有一個重聚的機會─那就是三大行星運行至銀河中央的那一天─「幻之日」，這一天會產生奇異的「星之門」，可以使三地人民到彼此的星球拜訪，進而產生和好的機會，可是……","繼續看下一段。","稍後再看吧。","reply_03")
end
function reply_03()
	if GetChoice()==0 then
		dia_04()
	else
		QuestEnd()
	end
end

function dia_04()
	Say("n1就在1000年前幻之日的時刻，古羅斯有人發明出人造的星之門，打開了通往其他行星的通道後，邪惡的亞斯人便帶著邪神士兵─魔機兵，大舉進攻加米蘭特。")
	Say("n1長達30年的星之戰爭就此展開，在星之戰初期，加米蘭特的人族和米古族處於相當的劣勢中，於戰事中節節敗退。")
	Say("n1在亞斯大軍即將取得完全勝利的時候，人族在聖地得到大地神─佳雅的指引，發現了「機動靈」─一種有意志的機械生命體，從此使戰爭進入了膠著狀態。")
	Say("n1長久的戰爭使得美麗的加米蘭特受到嚴重的環境破壞，讓原本可居住的地方愈來愈少。")
	Say("n1在一次事件中，善良的米古族人─尼尼亞救了一批被同伴遺棄的亞斯人，這份善意使得一些亞斯人有了改變，進而得到這些亞斯人的協助，")
	Say("n1學會使用人造星之門的技術，也了解到星之門將因星球的距離拉遠而不能發揮作用。")
	Say("n1在星之戰爭第29年，加米蘭特的人民決定運用星之門離開已不適合居住的加米蘭特星，全民逃難到原始的星球─尼路亞，")
	Say("n1可是這次的逃難，卻只有大約一半的人去到了尼路亞，留下的人則成了亞斯人的奴隸。")
	Say("n1而逃難的人，在尼路亞展開全新的生活，但他們也沒有忘記在加米蘭特的同胞，所以在尼路亞積極發展武鬥術與魔法。此時，尼尼亞成為了新國家的國王。")
	Say("n1經過5百多年後，在發展武鬥術和魔法最有成就的分別是白鷹騎士訓練學校和德爾莫亞魔法學園。")
	Say("n1在星之戰爭後9百多年的今天，幻之日即將再度來臨，人們為了解放在加米蘭特的同胞，都以成為兩所學校的學生而努力……","End")
end
