--星之門建造場守衛(MS004_A)

--MS004_Task:
-- 1: quest 

function start()
	if GetCountItemCount(10263)>=1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function dialogue_01()
	Say("n1星之門建造場是重要的地方。閒雜人等，一概不准進入！","End")
end

--玩家手持星之門通行証(10263)
function dialogue_02()
	Ask("n請出示星之門通行証。","出示星之門通行証。","守衛的工作真是辛苦呢！","reply_02")
end

function dialogue_02a()
	Say("n1"..get_title_name().."的"..GetName().."進入星之門建造場，現在已經幫你進行登記程序，請稍等。")
	Say("n1．．．")
	Say("n1．．．")
	Say("n1已經完成了登記程序，你可以進入星之門建造場了。","go_away")
end

function go_away()
	MovePlayer(7360,7200,15)
	QuestEnd()
end

function dialogue_02b()
	Say("n1星之門建造場是重要的地方。閒雜人等，快點離開吧！","End")
end

function reply_02()
	if GetChoice()==0 then
		dialogue_02a()
	else
		dialogue_02b()
	end
end

function get_title_name()
	careerType = GetData("Career")
	if careerType<=3 then
		return "格靈騎士團"
	elseif careerType==4 or careerType==8 then
		return "絲蘭狙擊兵團"
	elseif careerType<=7 then
		return "古尼爾魔法軍團"
	elseif careerType==10 then
		return "皇家精靈戰團"
	else
		return "尼尼亞補給團"
	end
end
