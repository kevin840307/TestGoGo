--Shop24 藥局(尼‧機)

function start()
	dialogue_Welcome()
end

function dialogue_Welcome()
	Ask("s歡迎光臨藥局，這裡有各式各樣的藥物。出門旅行，總要買一點來傍身吧？","讓我看看吧。","不要了。","reply_Welcome")
end

function reply_Welcome()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Career")==9 then
			OpenShop (24,80)
		else
			OpenShop (24,100)
		end
	else
		Say("s歡迎下次再光臨。","End")
	end
end
