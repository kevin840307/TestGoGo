function start()
	if GetFlag("TW008_C_Step") >= 2 then
		if GetFlag("TW008_D_Step") >=2 then
			if GetFlag("TW008_D_Step") >=4 then
				if GetFlag("TW008_D_Step") >=6 then
					if GetFlag("TW008_D_Step") >=7 then
						Step08()
					else	
						Step07()
					end
				else
					Step06()
				end
			else	
				Step04()
			end
		else
			Step02()
		end
	else
		Step01()	
	end
end

function Step01()
	Say("n1真心先生啊，人家好想念你啊．．．")
end

function Step02()
--<After TW008_C Step 02>
--<玩家沒有３個隊員在附近做出示愛的表情＞→Step03
	hasmember, teammember = GetTeamList_First()
	total = 0
	while hasmember == 1 do
		if GetTargetEmotion(teammember) == 11 then
			total = total + 1
		end
		hasmember, teammember = GetTeamList_Next()
	end
	if total < 3 then
		Step03()
	else
		Say("p1真心小姐～我是．．．")
		Say("n3你在幹什麼！？我已然有心上人了，快走開！！")
		Say("p6她好像很生氣的樣子．．．還是晚一點再來～")
		SetFlag("TW008_D_Step",2)
		SetFlag("TW008_Angry", GetRealTime())
	end
end

function Step03()

function Step07()

--<玩家沒有３個隊員在附近做出示愛的表情＞→Step03

		Step05()
