--TQ031_A	灼熱騎士團團長

--TQ031_Task
--1. lv > 50 and talked with TQ031_A
--2. talked with TQ031_B, need to get the 熱血之花
--3. talked with TQ031_A, need to get the 熱血之花 again
--4. quest completed

#include "CommComplex.txt"

function start()
	--＜完成TQ035之後＞
	--＜Reset TQ031＞→Step01
	--if GetFlag("TQ035_Done")==1 and GetFlag("TQ031_Done")==1 then
	--	RemoveFlag("TQ031_Task")
	--	RemoveFlag("TQ031_C")
	--	RemoveFlag("TQ031_Done")
	--end

	state = GetFlag("TQ031_Task")

	if state>=4 then
		dia_09()
	elseif state>=3 then
		equip, color = GetComplexArray()
		if CheckItem(10634)>0 then
			dia_07()
		elseif CheckItem(10633)>0 and equip==1 and color==3 then
			dia_06()
		else
			dia_05()
		end
	elseif state >= 2 then
		if CheckItem(10633)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif GetData("Level")>=50 then
		dia_01()
	else
		dia_02()
	end
end

function reply_06()
	if GetChoice()==0 then
		dia_06_a()
	end
	QuestEnd()
end

function dia_06_a()
--＜傳送玩家至(X,Y,Z)＞
	MovePlayer(3392,2592,60)
	Say("p7服下熱血之花保持鬥志。","End")
	RemoveItem(10633)
end

--＜玩家身上持有烈火岩(10634)＞
--＜玩家身上裝備空間少於1＞→Step09
function dia_07()
	if ItemListCountLeft(1)<1 then
		dia_08()
	else
		Say("n1啊！這真的是烈火岩！！")
		Say("n1太感謝你了！我就將這烈火岩之劍送給你吧！")
		Say("p7得到烈火岩之劍")
		Say("p7得到30000Exp","End")
		RemoveItem(10634)
		AddItem(10635)
		gainexp(30000)
		SetFlag("TQ031_Task", 4)
		SetFlag("TQ031_Done", 1)
	end
end

function dia_08()
	Say("p7身上沒有足夠的裝備空間啊．．．先整理一下好了。","End")
end

--＜After TQ031_A Step08＞
function dia_09()
	Say("n1感謝你幫我們完成採集烈火岩的任務！","End")
end

function gainexp(value)
	ModifyData("ExpUse", value)
	ModifyData("ExpAcc", value)
end
