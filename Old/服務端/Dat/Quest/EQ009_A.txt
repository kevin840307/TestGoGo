--//EQ009_A

function start()
	AddItem(10715)
	if GetWorldFlag("EQ009") <= 0 then
		Step01()
	elseif GetWorldFlag("EQ009") == 1 then
		Step03()
	elseif GetWorldFlag("EQ009") == 2 then
		Step05()
	elseif GetWorldFlag("EQ009") == 3 then
		Step07()
	elseif GetWorldFlag("EQ009") == 4 then
		Step09()
	elseif GetWorldFlag("EQ009") == 5 then
		Step11()
	elseif GetWorldFlag("EQ009") == 6 then
		Step13()
	end
end

--//＜如WorldFlag (EQ009= -1)＞
--//AddItem(10715) 
--//(P1)使用高級武道大會控制器！
--//(P)請選擇功能。
--//＜選擇＞
--//．開始領取參加表格。→Step02
--//．不需要了。→Quest End
function Step01()
	Say("p1使用高級武道大會控制器！")
	Ask("p請選擇功能。","開始領取參加表格。","查詢場地玩家人數。","不需要了。","Step01_Handler")
end

function Step01_Handler()
	if GetChoice() == 0 then
		Step02()
	elseif GetChoice() == 1 then
		Step15()
	else
		QuestEnd()
	end
end
