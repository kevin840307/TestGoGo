function start()
	init_data()
	if GetCountItemCount(item_index)<300 then
		Say("p7使用"..item_name.."...")
		dia_01()
	else
		Say("p7使用"..item_name.."...","dia_01")
	end
end
	
function dia_01()
	init_data()
	AddItem(item_index)

	if GetData("Career") ~= career_id or GetData("Job") ~= job_id then
		dia_02()
	elseif GetData("Level") < level_needed then
		dia_03()
	elseif HasSkill(pre_skill_index)~=1 then
		dia_04()
	elseif HasSkill(skill_index)==1 then
		dia_05()
	else
		Say("p7你的眼前出現一道白光，你看到了"..god_name.."的形象，這時祂對你說．．．")
		Ask("p你要學習"..skill_name.."嗎？", "是的。", "不是。", "reply_01")
	end
end

function reply_01()
	init_data()
	if GetChoice()==0 then
		dia_06()
	else
		dia_07()
	end
end

function dia_02()
	Say("p7只有真正的"..career_name.."才可學習"..skill_name.."。")
	QuestEnd()
end

function dia_03()
	Say("p7你的等級仍未可學習"..skill_name.."啊。")
	QuestEnd()
end

function dia_04()
	Say("p7要學習"..skill_name.."之前，你必須先學會"..pre_skill_name.."啊。")
	QuestEnd()
end

function dia_05()
	Say("p7你已經學會了"..skill_name.."。")
	QuestEnd()
end

function dia_06()
	Say("p7我現在就將"..skill_name.."傳授給你．．．")
	Say("p7你已經學會了"..skill_name.."。")
	RemoveItem(item_index)
	AddSkill(skill_index)
	flag_name = "SkillLearned_"..skill_index
	if HasSkill(skill_index)==1 then
		SetFlag(flag_name, 1)
	else
		SetFlag(flag_name, 0)
	end
	QuestEnd()
end

function dia_07()
	Say("p7需要學習"..skill_name.."時，可以再次召喚我。")
	QuestEnd()
end
