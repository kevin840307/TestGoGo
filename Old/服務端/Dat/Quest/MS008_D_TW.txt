function start()
	if GetFlag("MS008_D_Step") >= 2 then
		if GetFlag("MS008_D_Step")  >= 11 then
			Step14()
		else
			Step04()
		end
	else
		Step01()
	end
end

function Step11()

		Say("p1謝王子陛下。")
		Say("p7得到換領卷。")
		AddItem(10757)
		SetFlag("MS008_D_Step",11)
	end	
end

function go_next_question()
	next_step = GetFlag("MS008_Q")
	if next_step == 0 then
		Step05()
	elseif next_step == 1 then
		Step06()
	elseif next_step == 2 then
		Step07()
	elseif next_step == 3 then
		Step08()
	elseif next_step == 4 then
		Step09()
	elseif next_step == 5 then
		StepX()
	elseif next_step == 6 then
		QuestEnd()
	else
		QuestEnd()		
	end
end
