function start()
	--if GetFlag("TQ035_Done")==1 and GetFlag("TQ032_Done")==1 then
	--	RemoveFlag("TQ032_Task")
	--	RemoveFlag("TQ032_C")
	--	RemoveFlag("TQ032_C_Try")
	--	RemoveFlag("TQ032_Done")
	--end

	state = GetFlag("TQ032_Task")
	if state>=4 then
		dia_10()
	elseif state>=3 then
		equip, color = GetComplexArray()
		if CheckItem(10637)>0 then
			dia_08()
		elseif equip==1 and color==2 and CheckItem(10636)>0 then
			dia_06()
		else
			dia_05()
		end
	elseif state>=2 then
		if CheckItem(10636)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif GetData("Level")>=50 then
		dia_01()
	else
		dia_02()
	end
end

--＜玩家身上持有綠幽之氣(10637)＞
function dia_08()
--＜玩家身上裝備空間少於1＞→Step09
	if ItemListCountLeft(1)<1 then 
		dia_09()
	else
		Say("n1啊！這真的是綠幽之氣！！")
		Say("n1太感謝你了！我就將這綠幽下裝送給你吧！")
		Say("p7得到綠幽下裝")
		Say("p7得到30000Exp","End")
		RemoveItem(10637)
		AddItem(10638)
		gainexp(30000)
		SetFlag("TQ032_Task", 4)
		SetFlag("TQ032_Done", 1)
	end
end

function dia_09()
