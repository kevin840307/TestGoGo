function start()
	if GetFlag("TQ041_B_Step") >= 1 then
		if GetFlag("TQ041_H_Step") >= 1 then
			if GetFlag("TQ041_B_Step") >= 3 then
				if GetFlag("TQ041_I_Step") >= 3 then
					if GetFlag("TQ041_I_Step") >= 5 then
						if GetFlag("TQ041_B_Step") >= 6 then
							if GetFlag("TQ041_K_Step") >= 1 then
								if GetFlag("TQ041_B_Step") >= 8 then
									Step09()
								else
									Step08()
								end
							else
								Step07()
							end											
						else
							Step06()							
						end					
					else
						Step05()
					end
				else
					Step04()	
				end
			else
				Step03()
			end			
		else
			Step02()	
		end
	else
		Step01()
	end
end

	Say("p1在下正是！")
	Say("s今天是我們完成星空之門重建工程的大日子，亦表示我們將再次展開與亞斯人的戰爭，現在國家需要正直、勇敢和忠誠的勇士進行第一輪探索星之道的任務。")
	Say("s你願意接下這項任務嗎？")
	Say("p1可以為國家效力，是我的光榮！")
	Say("s那麼你先去準備一下，之後去找祖加尼．馬奇，他會帶你進入星之道的了。")
	SetFlag("TQ041_B_Step", 1)
end

function Step08()

--//＜After TQ041_K．Step01＞
	Say("s如果這是神的旨意，我們唯有遵從吧！")
	career =  GetData("Career")
	if career == 1 or career == 2 or career == 3 or career == 9 then
		Say("s為答謝你的幫忙，這真實之劍就送結你吧！")
		Say("p3感謝皇上賞賜！！")
		Say("p7得到真實之劍。")
	elseif career == 5 or career == 6 or career == 7 or career == 10 then
		Say("s為答謝你的幫忙，這真實之杖就送結你吧！")
		Say("p3感謝皇上賞賜！！")
		Say("p7得到真實之杖。")
	elseif career == 4 or career == 8 then
		Say("s為答謝你的幫忙，這真實之銃就送結你吧！")
		Say("p3感謝皇上賞賜！！")
		Say("p7得到真實之銃。")
	end
	Say("s請你去查出奧米加的希望這東西的下落！")
	if career == 1 or career == 2 or career == 3 or career == 9 then
		AddItem(10688)
	elseif career == 5 or career == 6 or career == 7 or career == 10 then
		AddItem(10689)
	elseif career == 4 or career == 8 then
		AddItem(10690)
	end
	SetFlag("TQ041_B_Step", 8)
end
