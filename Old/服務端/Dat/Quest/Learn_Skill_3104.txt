--Learn_Skill_3104(藍氣流)

function start()
	Giveback()
	if GetData("Career") == 3 then
	CheckOther()
	else
	Say("p7只有聖劍士才可學習藍氣流。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習藍氣流啊。","End")
	elseif HasSkill(3104)==1 then
	Say("p7你已經學會了藍氣流。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了風神的形象，這時祂對你說．．．")
		Ask("p你要學習藍氣流嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將藍氣流傳授給你．．．")
		Say("p7你已經學會了藍氣流。","End")
		RemoveItem(10342)
		AddSkill(3104)
	else
	Say("p7需要學習藍氣流時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10342)
end
