function start()

        Say("n1這個世界的和平！現在就要你去守護了！！")

        status = GetFlag("NQ026_B_STEP")
        if status < 2 then
            Say("p7男子：『噢！沒時間了，我一定要回去．．．』")
            Say("p7護士小姐：『先生！你傷得這麼重，不可以四處走動！要休息多點！』")
            if GetData("Level") >= 40 then
                Say("p1這位先生，你先不要衝動，你有什麼緊要事，我可以幫助你嗎？")
                Say("n7這個男子在打量著你．．．")
                Say("n7男子：『看你也有一定的實力，你應該可以幫得上忙，我的名字叫鐵男．．．』")
                Say("p6鐵男！？好土的名字啊．．．")
                Say("n1事情是這樣的，我是魔裝機甲軍團的駕駛員，奉命去收拾近日在四周搗亂的巨大魔物，可是魔物太強了！！我和我的隊員．．．")

                Say("n1要消滅那些魔物，就一定要用魔裝機甲的力量！求你！一定要替我的同伴報仇！")
                Say("p1好！這個就包在我身上！")
                Say("p6我要先找到能夠使用魔裝機甲的同伴！")
                SetFlag("NQ026_B_STEP",2)
            end
        elseif status < 4 then
            if CheckItem(10523)==1 and CheckItem(10524)==1 and CheckItem(10525)==1 and CheckItem(10526)==1 then
                if ItemListCountLeft(0) >= 1 and ItemListCountLeft(1) > 4 then
                    Say("n1啊！我果然沒有看錯人，你真的把那4隻魔物打倒了！")
                    Say("n1另外我還有一件事要拜託你！請你加入我們的研究所，成為魔裝機甲軍團的成員！")
                    Say("p1這個嘛．．．我怕．．．")
                    Say("n1因為在這次與魔物的戰鬥中，我們失去了很多同伴，你既然有這樣的能力，為何不加入我們一起去維護這世界的和平！？")
                    Say("p1我真的有這個能力嗎？")
                    Say("n1只要你肯努力便可以！")
                    Say("p1好吧！就讓我們一起去維護這世界的和平！")
                    Say("n1好！這是我們的戰鬥服和研究所通行証，你去機械之城找魔裝機甲研究所的守衛，他會讓你進去的。")

                    Say("p7失去孤寂魔物之魂")
                    RemoveItem(10524)
                    Say("p7失去戰爭魔物之魂")
                    RemoveItem(10525)
                    Say("p7失去饑餓魔物之魂")
                    RemoveItem(10526)
                    Say("p7失去冰花魔物之魂")

                    Say("p7得到烈火戰鬥服‧右袖")
                    AddItem(10532)
                    Say("p7得到烈火戰鬥服‧上裝")
                    AddItem(10533)
                    Say("p7得到烈火戰鬥服‧下裝")
                    AddItem(10534)
                    Say("p7得到烈火戰鬥服‧頭盔")

                Say("p1好！這個就包在我身上！")

        else
            Say("n1只要你拿著研究所通行証去機械之城找魔裝機甲研究所的守衛，他會讓你進去研究所的了。")
        end
    end
end
