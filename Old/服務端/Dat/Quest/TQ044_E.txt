function start()
	Step01()
end

function Step01()
	if ItemListCountLeft(0) < 1  then
		Step02()
	else
		if GetFlag("TQ044_A_Step") == 2 and GetFlag("TQ044_E_Step") ~= 1 then
			Say("p7得到蝙蝠奶瓶。")
			AddItem(10759,1)
			SetFlag("TQ044_E_Step",1)
		end
	end
end

function Step02()
	Say("p1原來身上已沒有足夠的物品空間了．先整理一下好了")
end
