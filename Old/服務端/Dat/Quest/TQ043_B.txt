function start()
	if GetFlag("TQ043_A_Step") >= 3 then
		if GetFlag("TQ043_B_Step") >= 2 then
			if GetFlag("TQ043_A_Step")>=9 then
				if GetFlag("TQ043_B_Step") >= 4 then
					if GetFlag("TQ043_D_Step") >=9 then
						if GetFlag("TQ043_B_Step")>=6 then
							if GetFlag("TQ043_C_Step") >= 6 then
								Step08()
							else
						    	Step07()
						    end
						else
							Step06()
						end
					else
						Step05()
					end
				else
					Step04()
				end
			else
				Step03()
			end
		else		
			Step02()
		end
	else
		Step01()
	end
end

function Step01()
	Say("n1最近好像有什麼不妥當，看來我要加強收集情報了。")
end

function Step02()
--＜After TQ043_A．function Step03＞
	Say("n1『"..GetName().."』，為什麼你氣急敗壞的樣子，是不是發生了什麼事。")
	Say("p3將軍，大事不妙了，請快快封鎖天空島。")
	Say("n1要封鎖任何地區，只有國王及皇子才有這個權力我只是一個將軍，沒有下令封鎖的權力，是發生了什麼事？")
	Say("p1天空島中給魔物入侵了，而且聖獸亦會襲擊人民，已經有人民因此而傷亡。")
	Say("n1情況嚴重，這件事不能遲的，你快準備去見國王陛下。")
	SetFlag("TQ043_B_Step",2)
end

function Step03()
--＜After TQ043_B．function Step02＞
	Ask("n你是不是已經準備好去見國王陛下？","是的","不是","Step03_Handler")
--＜選項＞
--．是的→MovePlayer (11136,2944,88)
--．不是→Quest End
end

function Step03_Handler()
	if GetChoice() == 0 then
		SetFlag("TQ043_B_Step",3)
		MovePlayer(11136,2944,88)
	else
		QuestEnd()
	end
end
