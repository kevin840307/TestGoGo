-- MS003_Task
-- 0: ShowTitle
-- 1: quest start
-- 2: get the blood, go to get metal
-- 3: get the metal, not the passport yet
-- 4: get the passport

function start()
	status = GetFlag("MS003_Task")
	if GetFlag("MS001_Done")~=1 then
		dialogue_01()
	elseif status >= 4 then
		dialogue_09()
	elseif status >= 3 then
		dialogue_06()
	elseif status >= 2 then
		if RemoveItem(10261)==1 then
			dialogue_05()
		else
			dialogue_04()
		end
	elseif status >= 1 then
		if GetCountItemCount(10257)>=1 then
			dialogue_03()
		else
			dialogue_02b()
		end
	else
		Say("m#STORY_NAME_02A.spr#STORY_NAME_02B.spr#030.mp3#","dialogue_02")
	end
end

function reply_06()

	else

end

function dialogue_07()
	Say("n1這是星之門通行証，你之後就可以自由出入星之門的建造場。")
	if ItemListCountLeft(0)<1 then
		Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		Say("p7得到“星之門通行証”","End")
		AddItem(10263)
		SetFlag("MS003_Task", 4)
		SetFlag("MS003_Done", 1)
	end
end

function get_title_name()
	careerType = GetData("Career")
	if careerType<=3 then
		return "格靈騎士團"
	elseif careerType==4 or careerType==8 then
		return "絲蘭狙擊兵團"
	elseif careerType<=7 then
		return "古尼爾魔法軍團"
	elseif careerType==10 then
		return "皇家精靈戰團"
	else
		return "尼尼亞補給團"
	end
end
