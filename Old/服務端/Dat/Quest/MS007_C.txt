--占卜師．露露絲(MS007_C)

--MS007_Task
-- 1: quest started
-- 2: go to find lolosie
-- 3: get the flower, go to find prince
-- 4: found prince, got the water
-- 5: get the report
-- 6: give the report to prince

--MS007_Jimbok
-- 1: visited lolosie

function start()
	status = GetFlag("MS007_Task")
	if status==2 then
		dialogue_01()
	elseif status==3 then
		dialogue_02()
	elseif status==4 then
		dialogue_03()
	elseif GetFlag("MS007_Jimbok")~=1 then
		dialogue_04()
	else
		randAns = random(5)
		if randAns==1 then
			dialogue_05()
		elseif randAns==2 then
			dialogue_05b()
		elseif randAns==3 then
			dialogue_05c()
		elseif randAns==4 then
			dialogue_05d()
		else
			dialogue_05e()
		end
	end
end

function dialogue_01b()
	if ItemListCountLeft(0)<1 then
		Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		Say("p7得到“星晨花”","End")
		AddItem(10329)
		SetFlag("MS007_Task", 3)
	end
end

function reply_05()
	if GetChoice()==3 then
		QuestEnd()
	elseif GetData("Money")<10 then
		Say("n1很對不起啊！你身上好像沒有足夠的星幣。","End")
	else
		ModifyData("Money", -10)
		Say("p7失去10元")
		if GetChoice()==0 then
			dialogue_jim_1()
		elseif GetChoice()==1 then
			dialogue_jim_2()
		else
			dialogue_jim_3()
		end
	end
end

function dialogue_end()
	Say("n1大地的兒女"..GetName().."，你的占卜到此為止了，願佳雅與你同在。")
	randAns=random(4)
	if randAns==1 then
		Say("n1未來的路還在不斷的變化中，艾達的詩篇只能指引你的方向，但最終的命運還在你的手裡。")
	elseif randAns==2 then
		Say("n1艾達的詩篇只能指引你的方向，剩下來就要看你如何面對命運了。")
	elseif randAns==3 then
		Say("n1在命運之輪的指引下前進，你會如何面對呢？")
	else
		Say("n1占卜的真確，在於你的信念與心靈，若你不全心信賴，占卜多少次也是枉然。")
	end
	QuestEnd()
end
