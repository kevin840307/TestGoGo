--古洛尼校長( MS006_E)

function start()
	if GetCountItemCount(10300)>=1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function dialogue_01()
	Say("n1高強的魔法師不會隨便使用魔法，因他知道就算是一個很少的魔法，也會影響自然的平衡。","End")
end

--手持「給古加尼的信」
function dialogue_02()
	Say("n1你這麼匆忙，難度有什麼突發的事情了？")
	Say("p1大皇子叫我把這封信交到你的手上。")
	Say("n1啊！是嗎？馬上給我看看。","dialogue_02b")
end
function dialogue_02b()
	Say("p7失去“給古洛尼的信”")

	Say("n7古洛尼校長馬上把信細讀")
	Say("n1喔！．．．唉，原來如此。")
	Say("n1唉，害得皇子也為我們擔心．．．")
	Say("p1校長，有什麼事嗎？")
	Say("n1我想你近日也會聽到一些留言，說騎士學校的學生正準備私下和我們的學生決鬥，這消息已傳到大皇子那兒了。")
	Say("n1我已呼籲大家不要和騎士學員衝突，我實在不想我兩兄弟的分歧，使兩所學園產生流血事件。")
	Say("p1兩兄弟？")
	Say("n1這裡很多人也知道，騎士學校的校長－古加尼校長是我的哥哥，我們為了證明自己的實力比對方強，分別行上不同的道路...")
	Say("n1卻想不到會影響到我們的學生，還使大皇子為我們憂心。")
	Say("n1這樣吧，請你盡快"..Color(0,45,198).."把此信交給大皇子"..Color(0,0,0).."，說明我會盡一切努力避免衝突。","dialogue_02c")
end
function dialogue_02c()
	Say("p7得到“古洛尼給大皇子的信”", "End")
	RemoveItem(10300)
	AddItem(10302)
end
