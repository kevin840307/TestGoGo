--Shop19+20 手套小屋

function start()
	dialogue_Welcome()
end

function dialogue_Welcome()
	Ask("s歡迎光臨手套小屋，這裡有各式各樣的手套裝備，要看看嗎？","我要看左手裝備。","我要看右手裝備。","不要了。","reply_Welcome")
end

function reply_Welcome()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Career")==9 then
			OpenShop (19,80)
		else
			OpenShop (19,100)
		end
	elseif choice == 1 then
		if GetData("Career")==9 then
			OpenShop (20,80)
		else
			OpenShop (20,100)
		end
	else
		Say("s歡迎下次再光臨。","End")
	end
end
