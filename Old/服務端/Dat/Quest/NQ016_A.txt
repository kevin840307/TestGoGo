--NQ016_A 穆蘭卡博士

--status (NQ016_Task)
-- 1: quest start, go to find 艾伯博士
-- 2: go items for 艾伯博士
-- 3: go to find 穆蘭卡博士
-- 4: quest completed

--NQ016_Done
-- 1: Quest completed, this flag will be checked by NQ017_A

function start()
	status = GetFlag("NQ016_Task")
	if status==4 then
		dialogue_05()
	elseif status==3 and GetCountItemCount(10216)>0 then
		dialogue_04()
	elseif status>= 1 then
		dialogue_03()
	elseif GetData("Level")<20 then
		dialogue_00()
	else
		dialogue_01()
	end
end

function dialogue_00()
	Say("n1靜一點喔！","End")
end

function dialogue_04()

	end

end

function dialogue_04b()
	RemoveItem(10216, 1)
	Say("p7得到2000EXP","End")
	SetFlag("NQ016_Task", 4)
	SetFlag("NQ016_Done", 1)
	gainexp(2000)
	AddItem(6001)
	if CheckItem(6001)>0 then
		SetFlag("NQ016_GetHorse", 1)
	else
		SetFlag("NQ016_GetHorse", 0)
	end
end

function dialogue_05()
	Say("n1試過用機甲了嗎？這是很好玩的。")
	Say("n1還有忘了告訴你一件事，當駕駛機甲後是不能移隨機甲裝備的，必須回收後才能裝配你的機甲。","End")
end

function reply_01()
	if GetChoice()==0 then
		dialogue_02a()
	else
		dialogue_02b()
	end
end

function gainexp(exp)

	ModifyData("ExpAcc",exp)
	ModifyData("ExpUse",exp)

end
