--Learn_Skill_4103(鐵彈擊)

function start()
	Giveback()
	if GetData("Career") == 4 then
	CheckOther()
	else
	Say("p7只有銃戰士才可學習鐵彈擊。","End")
	end
end

function CheckOther()
	if GetData("Level") < 20 then
	Say("p7你的等級仍未可學習鐵彈擊啊。","End")
	elseif HasSkill(4103)==1 then
	Say("p7你已經學會了鐵彈擊。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了雷神的形象，這時祂對你說．．．")
		Ask("p你要學習鐵彈擊嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將鐵彈擊傳授給你．．．")
		Say("p7你已經學會了鐵彈擊。","End")
		RemoveItem(10335)
		AddSkill(4103)
	else
	Say("p7需要學習鐵彈擊時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10335)
end
