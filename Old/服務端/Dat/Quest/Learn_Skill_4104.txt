--Learn_Skill_4104(閃影術)

function start()
	Giveback()
	if GetData("Career") == 4 then
	CheckOther()
	else
	Say("p7只有銃戰士才可學習閃影術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 30 then
	Say("p7你的等級仍未可學習閃影術啊。","End")
	elseif HasSkill(4104)==1 then
	Say("p7你已經學會了閃影術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了雷神的形象，這時祂對你說．．．")
		Ask("p你要學習閃影術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將閃影術傳授給你．．．")
		Say("p7你已經學會了閃影術。","End")
		RemoveItem(10343)
		AddSkill(4104)
	else
	Say("p7需要學習閃影術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10343)
end
