--Learn_Skill_9202(強化劑製造)

function start()
	Giveback()
	if GetData("Career") == 9 then
	CheckOther()
	else
	Say("p7只有道具使才可學習強化劑製造。","End")
	end
end

function CheckOther()
	if GetData("Level") < 40 then
	Say("p7你的等級仍未可學習強化劑製造啊。","End")
	elseif HasSkill(9202)==1 then
	Say("p7你已經學會了強化劑製造。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了創造之神的形象，這時祂對你說．．．")
		Ask("p你要學習強化劑製造嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將強化劑製造傳授給你．．．")
		Say("p7你已經學會了強化劑製造。","End")
		RemoveItem(10349)
		AddSkill(9202)
	else
	Say("p7需要學習強化劑製造時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10349)
end
