--TQ001_B 卡卡鳥

function start()
	lv = GetData("Level")
	kill = GetFlag("TQ001_Egg")
	if lv > 5 then
		QuestEnd()
	else
		if kill < 0 then
			kill=0
		end
		kill = kill + 1
		SetFlag("TQ001_Egg", kill)
		Say("p7你已經打倒了"..kill.."隻"..Color(0,45,198).."卡卡鳥"..Color(0,0,0).."．．．","CheckEnough")
	end
end

function CheckEnough()
	if GetFlag("TQ001_Egg") >= 5 then
		if ItemListCountLeft(0)<1 then
			Say("p7對不起，你身上的物品太多了，請清除一些物品後再來。","End")
		else
			Say("p7得到"..Color(0,45,198).."卡卡鳥蛋","End")
			SetFlag("TQ001_Egg",0)
			AddItem(2004,1)
		end
	else 
		QuestEnd()
	end
end
