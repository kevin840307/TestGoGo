--Shop29 打鐵舖(機)

function start()
	dialogue_Welcome()
end

function dialogue_Welcome()
	Ask("s歡迎光臨打鐵舖，這裡銃砲的款式就最齊全了。來看看吧！","讓我看看吧。","不要了。","reply_Welcome")
end

function reply_Welcome()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Career")==9 then
			OpenShop (29,80)
		else
			OpenShop (29,100)
		end
	else
		Say("s歡迎下次再光臨。","End")
	end
end
