--爆破機械偶‧改(MS004_L)

--MS004_Task
-- 1: quest start, go to 泰雲
-- 2: go to 依加魯穩定器
-- 3: go to 迪爾
-- 4: go to 資訊大樓
-- 5: go to 嘉迪尼事務長
-- 6: go to 格靈森林
-- 7: got the 野花電子卡
-- 8: 基地destoryed
-- 9: quest completed

function start()
	total = GetCountItemCount(10274)
	if GetFlag("MS004_Task")>=7 then
		QuestEnd()
	elseif total>=30 then
		dialogue_03()
	elseif total>=1 then
		dialogue_02()
	else
		dialogue_01()
	end
end

--玩家身上0支爆破穩定劑
function dialogue_01()
	Say("p7爆破機械偶‧改身上閃著紅光！")
	Say("p3轟隆隆～","dialogue_01a")
end

function dialogue_01a()
	Say("p7爆破機械偶‧改的爆炸對你做成傷害！")
	SetData("HP", 1)
end

--玩家身上1-29支爆破穩定劑
function dialogue_02()
	Say("p7爆破機械偶‧改身上閃著紅光！")
	Say("p7向爆破機械偶使用爆破穩定劑。")
	Say("p7穩定劑的數量好像不夠．．．")
	Say("p3轟隆隆～","dialogue_02a")
end
function dialogue_02a()
	Say("p7爆破機械偶‧改的爆炸對你做成傷害！")
	Say("p7失去所有爆破穩定劑","End")
	RemoveItem(10274, GetCountItemCount(10274))
	SetData("HP", 1)
end

--玩家手持30支爆破穩定劑
function dialogue_03()
	Say("p7爆破機械偶‧改身上閃著紅光！")
	Say("p7向爆破機械偶使用爆破穩定劑。")
	Say("p7阻止了爆破機械偶‧改的爆炸！")
	Say("p7失去所有爆破穩定劑")
	Say("p7得到“野花電子卡”","End")
	
	RemoveItem(10274, GetCountItemCount(10274))
	AddItem(10275)
	SetFlag("MS004_Task", 7)
end
