--Learn_Skill_10108(召靈魔法)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習召靈魔法。","End")
	end
end

function CheckOther()
	if GetData("Level") < 45 then
	Say("p7你的等級仍未可學習召靈魔法啊。","End")
	elseif HasSkill(10108)==1 then
	Say("p7你已經學會了召靈魔法。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習召靈魔法嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將召靈魔法傳授給你．．．")
		Say("p7你已經學會了召靈魔法。","End")
		RemoveItem(10359)
		AddSkill(10108)
	else
	Say("p7需要學習召靈魔法時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10359)
end
