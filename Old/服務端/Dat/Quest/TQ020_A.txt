--電遊人第九號當鋪(TQ020A)

function start()
	if GetCountItemCount(10442)>0 then
		dia_01()
	else
		dia_09()
	end
end

function dia_01()
	Ask("n歡迎！這裡是電遊人第九號當鋪，如果你找到以下物品，可能跟我換取不同的強化水。","30個波奇鳥羽毛。","30個波奇鳥甲殼。","30個機動偶靈件。","其他選擇。","reply_01")
end
function reply_01()
	ans = GetChoice()
	if ans==0 then
		dia_03()
	elseif ans==1 then
		dia_04()
	elseif ans==2 then
		dia_05()
	else
		dia_02()
	end
end

function dia_02()
	Ask("n如果你找到以下物品，可能跟我換取不同的強化水。","30個索爾魔怪毛皮。","30個索爾魔怪甲殼。","其他選擇。","還是不要了。","reply_02")
end
function reply_02()
	ans = GetChoice()
	if ans==0 then
		dia_06()
	elseif ans==1 then
		dia_07()
	elseif ans==2 then
		dia_01()
	else
		QuestEnd()
	end
end

function dia_03()
	if GetCountItemCount(10060)<30 then
		dia_08()
	else
		Say("n1這5枝生命強化水是給你的。")
		Say("n1交出30個波奇鳥羽毛。","dia_03b")
	end
end
function dia_03b()
	if ItemListCountLeft(0)<1 then
		Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		Say("p7得到5枝生命強化水。","remove_letter")
		RemoveItem(10060,30)
		AddItem(220,5)
	end
end
