--大亂鬥管理員 (TQ039_A)

function start()
	Ask("n歡迎你來到這裡，你想參加星空大亂鬥嗎？","我現在就要參加星空大亂鬥。","我要知道更多星空大亂鬥的資料。","我只是路過的．．．","reply_01")
end
function reply_01()
	ans = GetChoice()
	if ans==0 then
		dia_02()
	elseif ans==1 then
		dia_03()
	else
		QuestEnd()
	end
end

function dia_02()
	Say("n1進入混戰區的預備區之後，經過混戰區入口便可進入混戰區中。路上小心啊～","move")
end
function move()
	MovePlayer(7981,13642,67)
	QuestEnd()
end

function dia_03()
	Say("n1現在就讓我來說明一下星空大亂鬥的基本規則．．．")
	Say("n1經由我這裡便可進入混戰區中的預備區中。")
	Say("n1在預備區中經過入口便可進入混戰區或回到這裡。")
	Say("n1如你是使用機甲中、合體中、組隊中或召喚戰鬥精靈中是不能進入混戰區的。")
	Say("n1在混戰區中，你只可使用回復劑、強化劑和攻擊物品。")
	Say("n1而其他任務物品如古可的傳送蘑菇、技能捲軸等則不能使用。")
	Say("n1在混戰區中，玩家不可使用機甲、不可合體、不可組隊、不可召喚戰鬥精靈。")
	Say("n1在混戰區中死亡是不會增加死亡經驗值的。")
	Say("n1死亡後按Esc便會在復活點上復活。")
	Say("n1在混戰區中不設出口，如要離開混戰區，重新登入角色便可。")
	Say("n1角色在混戰區重新登入之後，便會回到這裡。")
	QuestEnd()
end
