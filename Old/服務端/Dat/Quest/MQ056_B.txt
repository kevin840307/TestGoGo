function start()
	if GetData("Level")<40 then
		dialogue_Below40()
	else
		switch_After40()
	end
end

function switch_After40()
	career=GetData("Career")
	if career==5 then
		status=GetFlag("MQ056_Task")
	elseif career==6 then
		status=GetFlag("MQ057_Task")
	elseif career==7 then
		status=GetFlag("MQ058_Task")
	else
		status=GetFlag("MQ059_Task")
	end

	if status>=3 then
		dialogue_final()
	elseif status==2 then
		dialogue_02()
	elseif status==1 then
		dialogue_01()
	else
		dialogue_Below40()
	end
end

function dialogue_Below40()
	Say("n1……")
	Say("p1這個女人神秘兮兮的……","End")
end

function dialogue_final()
	Say("n1……這世界，幾乎令我瘋狂起來。","End")
end

function dialogue_01()
	Say("n3……你終於來了，我等你好久了。")
	Say("p1你為什麼知道我會來？你就是巫婆？")
	Say("n1……真神已經在等著你了。")
	Say("p1你在說什麼？")
	Say("n1……你還不知道嗎？\n我需要你的力量。")
	Say("n1……你必須想起你來這兒是為了什麼。")
	Say("p1我來就是為了向你請教魔法和想和你談談關於你的事啊！")
	Say("n3……這跟本不重要。")

	elseif career==6 then
		dialogue_01_6()
	elseif career==7 then
		dialogue_01_7()
	else
		dialogue_01_8()
	end
end

function dialogue_02()
	if ItemListCountLeft(0)<2 then
		Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
	else
		career=GetData("Career")
		if career==5 then
			if RemoveItem(10239)==1 then
				dialogue_02_5a()
			else
				dialogue_02_5()
			end
		elseif career==6 then
			if RemoveItem(10243)==1 then
				dialogue_02_6a()
			else
				dialogue_02_6()
			end
		elseif career==7 then
			if RemoveItem(10246)==1 then
				dialogue_02_7a()
			else
				dialogue_02_7()
			end
		else
			if RemoveItem(10249)==1 then
				dialogue_02_8a()
			else
				dialogue_02_8()
			end
		end
	end
end
