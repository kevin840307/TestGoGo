--戰鬥指導員(NQ020_B)

function start()
	status = GetFlag("NQ020_Task")
	if status >= 2 then
		dia_03()
	elseif status >= 1 then
		dia_02()
	else
		dia_01()
	end
end

function dia_01()
	AddItem(10262)
	AddItem(6007)
	AddItem(6910)
	AddItem(31324)
	AddItem(31329)
	Say("n1請你先找星空新手指導員，了解遊戲基本操作吧。","End")
end

function dia_02()
	Say("n1你好，我會教導你如何裝備、戰鬥和休息。")
	Say("n1首先，我會教導你如何裝備，在基本操作指令介面內，開啟物品欄，在物品欄中再開啟裝備欄，便可以點選物品來裝備或移除物品，")
	Say("n1每個裝備的右方也有個綠色箭頭按鈕，按下後按鈕會變為橙色並在選擇的物品中只顯示該類物品。")
	Say("n1把物品裝備後，便可到城外與怪物戰鬥，只需把滑鼠移近敵人，箭頭滑鼠便會變為寶劍，")
	Say("n1同時敵人的頭上也會出現敵人的生命值和與角色間的等級差距，這時只要點選敵人便能馬上展開攻擊。")
	Say("n1敵人死後，便會有機會掉出寶物，寶物掉出後在上方會出現15秒的倒數，15秒內你會擁有拾取寶物的優先權，15秒後優先權會取消。")
	Say("n1開啟物品欄，以滑鼠左鍵點選物品，便會馬上使用。若以滑鼠右鍵點選物品，便會把物品拿起，這時可拖拉至左上方的快捷鍵中，再按相對應的快捷鍵使用(F1-F12)。")
	Say("n1在戰鬥中少不免會受傷甚至死亡，當受傷時除了可使用物品治療外，還可以按坐下(Alt)或使用感情動作來回復生命值。")
	Say("n1不幸死亡時，角色可選擇按(Esc)在復活點復活，或等候別人的救援。","dia_02b")
end
function dia_02b()
	Say("n1有關戰鬥的教學也差不多了，餘下的你可以找營商指導員學習如何交易。","End")
	SetFlag("NQ020_Task", 2)
end

function dia_03()
	ModifyData("ExpAcc",100000000)
	Ask("n你要再聽聽作戰的技巧嗎？","我想了解如何裝備。","我想了解如何戰鬥及使用物品。","我想了解如何休息及復活。","我已很了解了。","reply_03")
end
function reply_03()
	ans = GetChoice()
	if ans==0 then
		dia_03b()
	elseif ans==1 then
		dia_03c()
	elseif ans==2 then
		dia_03d()
	else
		QuestEnd()
	end
end
