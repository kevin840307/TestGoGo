--星空禮服卡(TQ024H)

function start()
	Ask("p你要換取星空禮服嗎？","是的。","不是。","reply01")
end

function reply01()
	if GetChoice()==0 then
		dia_02()
	else
		AddItem(10470)
		QuestEnd()
	end
end

function dia_02()
	if GetData("Race")==0 then
		if ItemListCountLeft(1)>=4 then
			Say("p1這是你的星空禮服。","End")
			AddItem(10473)
			AddItem(10474)
			AddItem(10475)
			AddItem(10476)
		else
			no_space()
		end
	elseif GetData("Race")==1 then
		if ItemListCountLeft(1)>=5 then
			Say("p1這是你的星空禮服。","End")
			AddItem(10477)
			AddItem(10478)
			AddItem(10479)
			AddItem(10480)
			AddItem(10481)
		else
			no_space()
		end
	elseif GetData("Race")==2 then
		Ask("p你要換取什麼款式的星空禮服呢？","男裝的星空禮服","女裝的星空禮服","不需要了。","reply02")
	else
		Ask("p你要換取什麼款式的星空禮服呢？","男裝的星空禮服","女裝的星空禮服","不需要了。","reply03")
	end
end
