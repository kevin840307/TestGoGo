function start()
	if GetFlag("MS007_Done") ~= 1 then
		Step01()
	else
		if GetFlag("MS008_A_Step") >= 4 then
			if GetFlag("MS008_A_Step") >= 6 then
				if GetFlag("MS008_B_Step") >= 2 then
					if GetFlag("MS008_A_Step") >= 9 then
						Step10()
					else
						Step09()
					end
				else
					Step08()
				end
			else
				Step05()
			end				
		else
			if GetFlag("MS008_A_Step") >= 0 then
				Step02()
			else
				show_title()
			end
		end
	end
	
end

function Step01()
--<玩家未完成主線任務第七章 MS007_Done>
	Say("n1看你這麼年輕，看來你還不會聽得懂那個傳說，小孩子，多去歷練吧。")
end
function show_title()
	SetFlag("MS008_A_Step", 0)
	Say("m#STORY_NAME_07A.spr#STORY_NAME_07B.spr#030.mp3#","Step02")
end
function Step02()
--＜如玩家已完成主線任務第七章＞
	Say("n1小孩子，你有見過立普嗎？")
	Say("p1怎麼沒有，花花立普、水藍立普這些我殺了一大堆，別小看我。")
	Say("n1若果你知道那個傳說，你便不會再忍心傷害立普們。")
	Say("p1立普先攻擊我的，我很是自衛才將牠們殺掉，這樣做有什麼不對。")
	Ask("n是因為你還未聽說過你個傳說吧，你想聽嗎？","我正忙著段練自己，下次吧。","反正我有的時間，就和你聊一下。","Step02_Handler")
--＜選項＞
--．我正忙著段練自己，下次吧。→Quest End
--．反正我有的時間，就和你聊一下。→Step03
end

function Step02_Handler()
	if GetChoice() == 0 then
		QuestEnd()
	else
		Step03()
	end
end

	Say("n5亞魯斯和佳雅為保護地上的人類而與卡奧斯展開了神之戰爭。在戰爭中，卡奧斯用力量迷惑不少人類成為了祂的戰士。")
	Say("n5而被卡奧斯欺騙了的人類更砍下了亞魯斯一條手臂，我們的米古人就是在亞魯斯血中誕生。從此亞魯斯就不再相信了人類，就只剩下佳雅與卡奧斯作戰。")

	Say("n5亞魯斯為協助佳雅去戰勝卡奧斯，便用了自己最後的力量去創造一種生物代替人類，這就是『立普』。")
	Say("p3怎麼？立普是亞魯斯賜給我們的？")

---．我相信，只是覺得很驚訝吧。→Step06

		Step04()
	else
