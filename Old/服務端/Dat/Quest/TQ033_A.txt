function start()
	--if GetFlag("TQ035_Done")==1 and GetFlag("TQ033_Done")==1 then
	--	RemoveFlag("TQ033_Task")
	--	RemoveFlag("TQ033_C")
	--	RemoveFlag("TQ033_Done")
	--end

	state = GetFlag("TQ033_Task")

	if state>=4 then
		dia_10()
	elseif state>=3 then
		equip, color = GetComplexArray()
		if CheckItem(10640)>0 then
			dia_08()
		elseif CheckItem(10639)>0 and equip==1 and color==1 then
			dia_06()
		else
			dia_05()
		end
	elseif state >= 2 then
		if CheckItem(10639)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif GetData("Level")>=50 then
		dia_01()
	else
		dia_02()
	end
end

		dia_09()
	else
		Say("n1啊！這真的是烈火岩！！原來是黑岩魔龍獸幽靈作怪．．．")
		Say("n1太感謝你了！我就將這黑魔岩上裝送給你吧！")
		Say("p7得到黑魔岩上裝")
		Say("p7得到30000Exp","End")
		RemoveItem(10640)
		AddItem(10641)
		gainexp(30000)
		SetFlag("TQ033_Task", 4)
		SetFlag("TQ033_Done", 1)
	end
end
