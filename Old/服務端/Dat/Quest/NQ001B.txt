--NQ001_B 茶店小姐佩玲

function start()
	if GetFlag("NQ001_GetTea")==1 then
		Say("n1你不是要拿回去給伊寇先生嗎？", "End")
	elseif GetFlag("NQ_001_GetQuest")==1 then
		Say("n2歡迎來到森林茶店～！想要休息一下嗎？")
		Say("p1請問有沒有伊寇先生一向飲用的那種茶葉？")
		Say("n1啊！你說是帕特紅茶嗎？這種茶葉伊寇先生他真的很喜歡呢！")
		Say("p1那多少錢呢？")
		Ask("n嗯！打折後一共100元！","這是100元，請點收～","好貴喔！", "Step2")
	else
		Say("n2歡迎來到森林茶店～！想要休息一下嗎？")
		Say("p1不用了。", "End")
	end
end

function Step2()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Money") >= 100 then
			if ItemListCountLeft(0)<1 then
				Say("n1對不起，你身上的物品太多了，請清除一些物品後再來。","End")
			else
				AddItem (10008,1)
				ModifyData("Money", -100)
				SetFlag("NQ001_GetTea",1)
				Say("n1謝謝惠顧～有空再來唷！", "End")
			end
		else
			Say("n1你的錢未夠啊！這樣我會蝕本的！", "End")
		end
	else
		Say("n4可是已經打折了唷……", "End")
	end
end
