--Learn_Skill_10113(飼餵術)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習飼餵術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 13 then
	Say("p7你的等級仍未可學習飼餵術啊。","End")
	elseif HasSkill(10113)==1 then
	Say("p7你已經學會了飼餵術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習飼餵術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將飼餵術傳授給你．．．")
		Say("p7你已經學會了飼餵術。","End")
		RemoveItem(10364)
		AddSkill(10113)
	else
	Say("p7需要學習飼餵術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10364)
end
