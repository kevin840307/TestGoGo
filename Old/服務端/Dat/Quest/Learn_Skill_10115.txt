--Learn_Skill_10115(訓練術)

function start()
	Giveback()
	if GetData("Career") == 10 then
	CheckOther()
	else
	Say("p7只有精靈使才可學習訓練術。","End")
	end
end

function CheckOther()
	if GetData("Level") < 25 then
	Say("p7你的等級仍未可學習訓練術啊。","End")
	elseif HasSkill(10115)==1 then
	Say("p7你已經學會了訓練術。","End")
	else
	Say("p7你的眼前出現一道白光，你看到了精靈王的形象，這時祂對你說．．．")
		Ask("p你要學習訓練術嗎?","是的。","不是。","Teach")
	end
end

function Teach()
	choice = GetChoice()
	if choice == 0 then
		Say("p7我現在就將訓練術傳授給你．．．")
		Say("p7你已經學會了訓練術。","End")
		RemoveItem(10366)
		AddSkill(10115)
	else
	Say("p7需要學習訓練術時，可以再次召喚我。","End")
	end
end

function Giveback()
	AddItem(10366)
end
