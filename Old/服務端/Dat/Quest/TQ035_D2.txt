--特級軍餉券
--TQ035_D2

--world flag
--TQ035_D2_Bingo:last time player gain bingo

--flag
--TQ035_D2_EverBingo:1 if the player ever gain the bingo

function lGetItemName()
	return "特級軍餉券", 10664
end

function lGetRandGift()
	ans = random(100)
	if ans <= 30 then		--30%
		AddItem(252,5)
		return "加速藥x5", 252
	elseif ans <= 50 then	--20%
		AddItem(10235)
		return "榮譽勳章", 10235
	elseif ans <= 55 then	--5%
		AddItem(10575)
		return "封印山洞仙級團簽證", 10575
	elseif ans <= 65 then	--10%
		AddItem(10522)
		return "古代機甲之墓傳送器", 10522
	elseif ans <= 74 then	--9%
		AddItem(10592)
		return "特級天馬裝備換領券", 10592
	elseif ans <= 89 then	--15%
		AddItem(10307)
		return "小天使之翼", 10307
	elseif ans <= 99 then	--10%
		AddItem(10518)
		return "重星號機甲囊", 10518
	else					--1%
		nextTime = GetTimeForDay(7) + GetWorldFlag("TQ035_D2_Bingo")
		if GetFlag("TQ035_D2_EverBingo") == 1 then
			AddItem(10518)
			return "重星號機甲囊", 10518
		elseif GetRealTime() < nextTime then
			AddItem(10518)
			return "重星號機甲囊", 10518
		else
			SetFlag("TQ035_D2_EverBingo", 1)
			SetWorldFlag("TQ035_D2_Bingo", GetRealTime())
			gainexp(50000000)
			return "5000萬Exp", 0
		end
	end
end

function gainexp(value)
	ModifyData("ExpUse", value)
	ModifyData("ExpAcc", value)
end
