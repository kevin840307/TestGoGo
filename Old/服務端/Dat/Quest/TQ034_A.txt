function start()
	--if GetFlag("TQ035_Done")==1 and GetFlag("TQ034_Done")==1 then
	--	RemoveFlag("TQ034_Task")
	--	RemoveFlag("TQ034_Done")
	--end

	state = GetFlag("TQ034_Task")

	if state>=6 then
		dia_10()
	elseif state>=5 then
		equip, color = GetComplexArray()
		if CheckItem(10644)>0 then
			dia_08()
		elseif CheckItem(10642)>0 and equip==1 and color==4 then
			dia_06()
		else
			dia_05()
		end
	elseif state >= 3 then
		if CheckItem(10642)>0 then
			dia_04()
		else
			dia_03()
		end
	elseif GetData("Level")>=50 then
		dia_01()
	else
		dia_02()
	end
end

--＜玩家身上持有冰之依加魯結晶(10644)＞
function dia_08()
	if ItemListCountLeft(1)<2 then
		dia_09()
	else
		Say("n1啊！這真的是冰之依加魯結晶！！原來是冰窟魔鳥在作怪．．．")
		Say("n1太感謝你了！我就將這對寒流護手送給你吧！")
		Say("p7得到寒流左護手")
		Say("p7得到寒流右護手")
		Say("p7得到30000Exp","End")
		RemoveItem(10644)
		AddItem(10645)
		AddItem(10646)
		gainexp(30000)
		SetFlag("TQ034_Task",6)
		SetFlag("TQ034_Done",1)
	end
end
