--Shop27 打鐵舖(騎)

function start()
	dialogue_Welcome()
end

function dialogue_Welcome()
	Ask("s歡迎光臨打鐵舖，這裡劍的款式就最齊全的。來看看吧！","讓我看看吧。","不要了。","reply_Welcome")
end

function reply_Welcome()
	choice = GetChoice()
	if choice == 0 then
		if GetData("Career")==9 then
			OpenShop (27,80)
		else
			OpenShop (27,100)
		end
	else
		Say("s歡迎下次再光臨。","End")
	end
end
