--MS001_A 禁衛長柏魯達

--MS001_Task
-- 0: quest title shown
-- 1: quest started, go to find 50冰柏古獸的牙
-- 2: back from the tomb
-- 3: quest competed

--MS001_Done
-- 1: quest has been done once

function start()
	status = GetFlag("MS001_Task")
	if GetFlag("MS001_Task")==-1 then
		show_title()
	elseif GetFlag("MS001_Done")==1 then
		dialogue_05()
	elseif status==2 and CheckItem(10225)>0 then
		dialogue_04()
	elseif status>=1 then
		dialogue_03()
	elseif GetData("Career")~=0 then
		dialogue_02()
	else
		dialogue_01()
	end
end

function show_title()
	SetFlag("MS001_Task", 0)
	Say("m#STORY_NAME_00A.spr#STORY_NAME_00B.spr#030.mp3#","goCheckCareer")
end

function goCheckCareer()
	if GetData("Career")~=0 then
		dialogue_02()
	else
		dialogue_01()
	end
end

	Ask("n十分之好，看來你已把英雄之心帶來了。","對，我已把英雄之心帶回來了。","還是遲一會才給他。","reply_04")
end

function dialogue_04a()
	if ItemListCountLeft(1)<1 then
		Say("n1對不起，你身上的裝備太多了，請清除一些裝備後再來。","End")
	else
		Say("n1很好，以吾皇之名，"..GetName().."正式成為王國軍的一份子，今後你必須守護正義保衛國土，面對邪惡而不退縮。")
		Say("p7失去“英雄之心”")
		RemoveItem(10225)
		SetFlag("MS001_Task", 3)
		SetFlag("MS001_Done", 1)
		careerType = GetData("Career")
		if careerType<=3 then
			dialogue_04aa()
		elseif careerType==4 or careerType==8 then
			dialogue_04ab()
		elseif careerType<=7 then
			dialogue_04ac()
		elseif careerType==10 then
			dialogue_04ad()
		else
			dialogue_04ae()
		end
	end
end
